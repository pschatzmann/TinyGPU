11/**
12 * @brief Common base class for 8-bit parallel ("Intel 8080"-style) TFT
13 * display drivers: 8 data lines (D0-D7), WR (write strobe), RD (read
14 * strobe, unused - this class only writes), RS/DC (command vs. data) and
15 * CS. This is the bus MCUFRIEND-style shields and many 40-pin parallel
16 * TFT breakout boards use, wired straight to 8 GPIOs rather than through
17 * SPI/QSPI - no dedicated bus peripheral is assumed, every data line and
18 * the WR strobe is bit-banged via digitalWrite(), the same portable
19 * approach TinyGPU/Input/TouchDriverArduino.h's TouchDriverBitBang uses
20 * for a resistive touch panel's four GPIO pins. This trades throughput
21 * (one digitalWrite() per bit, per byte) for working unmodified on any
22 * Arduino core - a hardware-accelerated 8080 driver (e.g. ESP32's LCD_CAM/
23 * esp_lcd_panel_io_i80) would be faster but ties the implementation to one
24 * platform's peripheral, the same tradeoff DisplayDriverQSPI.h already
25 * makes explicitly for ESP32-only QSPI.
26 *
27 * The command/parameter protocol on the wire (0x2A/0x2B/0x2C column/row/
28 * RAM-write, 0x36 MADCTL, 0x3A pixel format, ...) is identical to the SPI
29 * wiring of the same controller families (see DisplayDriverSPI.h) - only
30 * how each command/data byte reaches the panel differs, so concrete
31 * drivers here mirror DisplayDriverSPI.h's ILI9341Driver/ST7789Driver/...
32 * init sequences byte-for-byte.
33 */
34template <typename RGB_T = RGB565>
35class DisplayDriverParallel8 : public DisplayDriver<RGB_T> {
36 public:
37 /// @param d0..d7 the 8 data lines, D0 = least significant bit.
38 /// @param wr write-strobe pin: data is latched by the panel on WR's
39 /// rising edge.
40 /// @param dc data/command ("RS") select pin: LOW selects command, HIGH
41 /// selects data - same polarity/role as DisplayDriverSPI's dc pin.
42 /// @param cs chip-select pin, or -1 if the panel has CS permanently
43 /// tied low (common on single-panel parallel shields).
44 /// @param rd read-strobe pin, or -1 if unused - this class never reads
45 /// from the panel, so it only needs to hold RD in its idle (HIGH,
46 /// inactive) state so the panel doesn't drive the (shared, bidirectional
47 /// on some boards) data bus back.

78 bool writeData(ISurface<RGB_T>& surface, size_t x, size_t y) override {
79 static_assert(sizeof(RGB_T) == 2,

81 "wire byte order");

83 beginTransaction();

85 const uint8_t* src = surface.data();
86 const size_t n = surface.size();

89 }
90 endTransaction();

113 void setupPinsAndReset() {
114 for (int8_t pin : d_) {
115 pinMode(pin, OUTPUT);
116 }
117 pinMode(wr_, OUTPUT);
118 digitalWrite(wr_, HIGH);
119 pinMode(dc_, OUTPUT);

127 }

161 /// Writes each byte as a separate writeData8() call - a convenience for
162 /// sending a command's multi-byte parameter list during init sequences,
163 /// matching DisplayDriverSPI::writeDataN().
164 void writeDataN(std::initializer_list<uint8_t> bytes) {
165 for (uint8_t b : bytes) {
166 writeData8(b);
167 }
168 }
169
170 private:
171 void beginTransaction() {
172 if (cs_ >= 0) digitalWrite(cs_, LOW);
173 }
174
175 void endTransaction() {
176 if (cs_ >= 0) digitalWrite(cs_, HIGH);
177 }
178
179 /// Drives one byte onto D0-D7 and pulses WR - the panel latches the bus
180 /// on WR's rising edge (LOW then HIGH).
181 void writeByte(uint8_t data) {
182 for (uint8_t i = 0; i < 8; ++i) {
183 digitalWrite(d_[i], (data >> i) & 0x01);
184 }
185 digitalWrite(wr_, LOW);
186 digitalWrite(wr_, HIGH);
187 }
