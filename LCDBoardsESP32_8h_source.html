2/**
3 * @file LCDBoardsESP32.h
4 * @brief One-call setup for ESP32(-S3) boards from the arduino-audio-tools
5 * "Audio Boards" wiki that combine an LCD/TFT display with a touch
6 * controller (and, on the same board, an I2S audio codec/amp):
7 * https://github.com/pschatzmann/arduino-audio-tools/wiki/Audio-Boards
8 *
9 * Bringing one of these boards up normally means hand-wiring the SPI/QSPI
10 * bus, the backlight pin, the display controller's init sequence, the
11 * touch controller's I2C bus, and separately tracking the board's I2S
12 * pins for the audio side - each board class here does the display/touch
13 * part in a single begin() call and exposes the I2S pin assignment so it
14 * can be handed to e.g. audio_tools::I2SConfig (TinyGPU itself has no
15 * audio-tools dependency, so I2SPins is a plain pin struct, not a
16 * driver).
17 *
18 * Only boards that actually have an LCD are covered - audio-only boards
19 * on that wiki page are out of scope for this graphics library. See
20 * LCDBoards.h for the platform-independent LCDBoard interface these
21 * classes implement.
22 */

36/**
37 * @brief "2.8" ESP32-S3 Display" - ILI9341 SPI TFT + FT6336G capacitive
38 * touch (I2C, shares its bus with the board's ES8311 audio codec).
39 *
40 * Display: ILI9341, 240x320, SPI - CS=10 DC=46 SCK=12 MOSI=11 MISO=13
41 * BL=45; RST is tied to the board's EN line, so there is no
42 * dedicated reset pin to drive. Needs display inversion turned
43 * on (done in begin()) or colors render as their photographic
44 * negative - confirmed on real hardware.
45 * Touch: FT6336G (register-compatible with TouchDriverFT6236), I2C -
46 * SDA=16 SCL=15 RST=18, physical IRQ=17. The IRQ pin is not
47 * wired into the driver (left at -1, forcing register polling)
48 * because this line's idle polarity was never confirmed on real
49 * hardware; TouchDriverFT6236::isTouched() would otherwise skip
50 * the I2C read whenever it reads the "wrong" level.
51 * I2S (ES8311 codec): MCLK=4 BCK=5 WS=7 DOUT=8 DIN=6. Speaker amp
52 * (FM8002E) enable: IO1, active low.
53 * LED: single WS2812-compatible RGB LED on IO42.
54 *
55 * @note Audio setup: this board's ES8311 codec needs its own I2C init (volume,
56 * mic gain, ...), not just I2S pins, so plain I2SStream
57 * leaves the codec unconfigured. If the sketch also depends on
58 * arduino-audio-driver, use its ready-made board definition instead,
59 * which owns the codec init and I2S pins together:
60 *
61 * #include "AudioTools.h"
62 * #include "AudioTools/AudioLibs/AudioBoardStream.h"
63 * #include "AudioBoards/ESP32S3HosyondDisplay.h" // arduino-audio-driver
64 *
65 * audio_tools::AudioBoardStream out(audio_driver::ESP32S3HosyondDisplay);
66 * auto cfg = out.defaultConfig(TX_MODE);
67 * out.begin(cfg);
68 * out.setVolume(0.5f);
69 *
70 * Without arduino-audio-driver, i2s()/setI2SPins() still give the raw
71 * MCLK/BCK/WS/DOUT/DIN pins for a plain I2SStream, but the caller is then
72 * responsible for the ES8311's I2C init and driving the PA-enable pin
73 * (i2s().paEnable, active low per paEnableActiveLow) itself.
74 */
75class LCDBoardESP32S3_2_8Display : public LCDBoard {
76 public:
77 /// Sets up the backlight, SPI bus, display controller (incl. inversion),
78 /// and touch controller. Returns false if the display or touch begin()
79 /// fails.
80 bool begin() override {
81 pinMode(kPinBacklight, OUTPUT);
82 digitalWrite(kPinBacklight, HIGH);
83
84 SPI.begin(kPinSck, kPinMiso, kPinMosi, kPinCs);
85 if (!display_.begin()) return false;
86 display_.setInvertColor(true);
87
88 Wire.begin(kPinTouchSda, kPinTouchScl);
89 return touch_.begin();
90 }
91
92 /// Panel width in pixels.
93 size_t width() const override { return 240; }
94 /// Panel height in pixels.
95 size_t height() const override { return 320; }
96
97 /// The board's display driver.
98 ILI9341Driver<RGB565>& display() override { return display_; }
99 /// The board's touch controller.
100 TouchDriver* touch() override { return &touch_; }
101 /// The board's I2S/PA pin assignment.
102 const I2SPins& i2s() const override { return i2s_; }
103 /// The board's LED pin assignment.
104 const LEDPins& led() const override { return led_; }
105
106 private:
107 static constexpr int8_t kPinCs = 10;
108 static constexpr int8_t kPinDc = 46;
109 static constexpr int8_t kPinSck = 12;
110 static constexpr int8_t kPinMosi = 11;
111 static constexpr int8_t kPinMiso = 13;
112 static constexpr int8_t kPinBacklight = 45;
113 static constexpr int8_t kPinTouchSda = 16;
114 static constexpr int8_t kPinTouchScl = 15;
115 static constexpr int8_t kPinTouchRst = 18;
116
117 // 80MHz (this panel's SPI clock ceiling on this board's wiring, verified
118 // on real hardware) instead of the constructor's 40MHz default - halves
119 // per-frame SPI transfer time versus leaving it at the default. Rotation
120 // stays kNone here since callers set it explicitly (e.g. via
121 // OutputTinyGPU::setRotation()) after begin().
122 ILI9341Driver<RGB565> display_{SPI, kPinCs, kPinDc, /*rst=*/-1,
123 ILI9341Driver<RGB565>::Rotation::kNone,
124 /*frequencyHz=*/80000000};
125 TouchDriverFT6236 touch_{Wire, kPinTouchRst, /*irq=*/-1};
126 I2SPins i2s_{/*mclk=*/4, /*bck=*/5, /*ws=*/7,
127 /*dataOut=*/8, /*dataIn=*/6,
128 /*paEnable=*/1, /*paEnableActiveLow=*/true};
129 LEDPins led_{/*single=*/42};
130};
