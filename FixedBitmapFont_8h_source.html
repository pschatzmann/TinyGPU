14/**
15 * @brief Fixed-size bitmap font backed by a caller-supplied glyph table.
16 *
17 * @tparam Width Glyph width in pixels.
18 * @tparam Height Glyph height in pixels.
19 * @tparam RowT Unsigned integer type used to store one glyph row. Only the
20 * `Width` most significant bits of each row are used (bit `Width-1` is the
21 * left-most pixel), mirroring the bit order used by BitmapFont.
22 * @tparam RGB_T The pixel color type. Can be RGB565, RGB666, RGB888, etc.
23 *
24 * Concrete fonts (e.g. converted from vendor font tables) subclass this and
25 * pass their own glyph table to the constructor. Only a contiguous ASCII
26 * range starting at `firstChar` is supported; any code point outside that
27 * range falls back to the glyph for '?'.
28 */
29template <uint8_t Width, uint8_t Height, typename RowT, typename RGB_T = RGB565>
30class FixedBitmapFont : public IFont<RGB_T> {
31 public:
32 /// Glyph storage: one row entry per pixel row.
33 using Glyph = std::array<RowT, Height>;
34 /// Unicode code point type used during UTF-8 decoding.
35 using CodePoint = uint32_t;
36
37 /// Width of a glyph in pixels.
38 static constexpr uint8_t kGlyphWidth = Width;
39 /// Height of a glyph in pixels.
40 static constexpr uint8_t kGlyphHeight = Height;
41
42 /// Creates a fixed bitmap font over a contiguous glyph table.
43 FixedBitmapFont(const Glyph* table, size_t glyphCount, char firstChar = ' ')
44 : table_(table), glyphCount_(glyphCount), firstChar_(firstChar) {}
45
46 /// Returns the glyph for an 8-bit character.
47 const Glyph& glyph(char character) const {
48 return glyph(static_cast<CodePoint>(static_cast<unsigned char>(character)));
49 }
50
51 /// Returns the glyph for a Unicode code point.
52 const Glyph& glyph(CodePoint codePoint) const {
53 const auto first = static_cast<unsigned char>(firstChar_);
54 if (codePoint >= first && codePoint < first + glyphCount_) {
55 return table_[codePoint - first];
56 }
57 return replacementGlyph();
58 }
59
60 /// Returns whether a pixel is set in a code point glyph.
61 bool pixel(CodePoint codePoint, uint8_t x, uint8_t y) const {
62 if (x >= kGlyphWidth || y >= kGlyphHeight) {
63 return false;
64 }
65 const RowT rowMask = glyph(codePoint)[y];
66 return (rowMask & static_cast<RowT>(RowT(1) << (kGlyphWidth - 1 - x))) != 0;
67 }

69 /// Draws a single Unicode code point.
70 void drawCodePoint(ISurface<RGB_T>& target, int16_t x, int16_t y,
71 CodePoint codePoint, RGB_T foreground,
72 RGB_T background = RGB_T(0), bool opaque = false,
73 uint8_t scale = 1) const {
74 if (scale == 0) {
75 scale = 1;
76 }
77
78 for (uint8_t row = 0; row < kGlyphHeight; ++row) {
79 for (uint8_t column = 0; column < kGlyphWidth; ++column) {
80 const int16_t pixelX = static_cast<int16_t>(x + (column * scale));
81 const int16_t pixelY = static_cast<int16_t>(y + (row * scale));
82
83 if (pixel(codePoint, column, row)) {
84 target.fillRect(pixelX, pixelY, scale, scale, foreground);
85 } else if (opaque) {
86 target.fillRect(pixelX, pixelY, scale, scale, background);
87 }
88 }
89 }
90 }

101 /// Draws a UTF-8 text string.
102 void drawText(ISurface<RGB_T>& target, int16_t x, int16_t y, const char* text,
103 RGB_T foreground, RGB_T background = RGB_T(0),
104 bool opaque = false, uint8_t scale = 1, uint8_t spacing = 1,
105 uint8_t lineSpacing = 1) const override {
106 if (text == nullptr) {
107 return;
108 }
109 if (scale == 0) {
110 scale = 1;
111 }
112
113 const int16_t advanceX =
114 static_cast<int16_t>((kGlyphWidth * scale) + spacing);
115 const int16_t advanceY =
116 static_cast<int16_t>((kGlyphHeight * scale) + lineSpacing);
117
118 int16_t cursorX = x;
119 int16_t cursorY = y;
120 const char* current = text;
121 while (*current != '\0') {
122 if (*current == '\n') {
123 ++current;
124 cursorX = x;
125 cursorY = static_cast<int16_t>(cursorY + advanceY);
126 continue;
127 }
128
129 const CodePoint codePoint = decodeNextUtf8(current);
130 drawCodePoint(target, cursorX, cursorY, codePoint, foreground, background,
131 opaque, scale);
132 cursorX = static_cast<int16_t>(cursorX + advanceX);
133 }
134 }

136 /// Returns the width of the longest text line in pixels.
137 size_t measureTextWidth(const char* text, uint8_t scale = 1,
138 uint8_t spacing = 1) const override {
139 if (text == nullptr || *text == '\0') {
140 return 0;
141 }
142 if (scale == 0) {
143 scale = 1;
144 }
145
146 size_t lineLength = 0;
147 size_t longestLine = 0;
148 const char* current = text;
149 while (*current != '\0') {
150 if (*current == '\n') {
151 ++current;
152 if (lineLength > longestLine) {
153 longestLine = lineLength;
154 }
155 lineLength = 0;
156 } else {
157 decodeNextUtf8(current);
158 ++lineLength;
159 }
160 }
161
162 if (lineLength > longestLine) {
163 longestLine = lineLength;
164 }
165
166 if (longestLine == 0) {
167 return 0;
168 }
169
170 return (longestLine * kGlyphWidth * scale) + ((longestLine - 1) * spacing);
171 }

174 size_t measureTextHeight(const char* text, uint8_t scale = 1,
175 uint8_t lineSpacing = 1) const override {
176 if (text == nullptr || *text == '\0') {
177 return 0;
178 }
179 if (scale == 0) {
180 scale = 1;
181 }
182
183 size_t lines = 1;
184 for (const char* current = text; *current != '\0'; ++current) {
185 if (*current == '\n') {
186 ++lines;
187 }
188 }
189
190 return (lines * kGlyphHeight * scale) + ((lines - 1) * lineSpacing);
191 }

194 size_t getHeight(uint8_t scale) const override {

203 const Glyph& replacementGlyph() const {
204 const auto first = static_cast<unsigned char>(firstChar_);
205 const auto question = static_cast<unsigned char>('?');
206 if (question >= first && question < first + glyphCount_) {
207 return table_[question - first];
208 }
209 return table_[0];
210 }
211
212 CodePoint decodeNextUtf8(const char*& current) const {
213 const unsigned char firstByte = static_cast<unsigned char>(*current);
214 if (firstByte == 0) {
215 return 0;
216 }
217
218 if ((firstByte & 0x80U) == 0) {
219 ++current;
220 return static_cast<CodePoint>(firstByte);
221 }
222
223 uint8_t sequenceLength = 0;
224 CodePoint codePoint = 0;
225 if ((firstByte & 0xE0U) == 0xC0U) {
226 sequenceLength = 2;
227 codePoint = static_cast<CodePoint>(firstByte & 0x1FU);
228 } else if ((firstByte & 0xF0U) == 0xE0U) {
229 sequenceLength = 3;
230 codePoint = static_cast<CodePoint>(firstByte & 0x0FU);
231 } else if ((firstByte & 0xF8U) == 0xF0U) {
232 sequenceLength = 4;
233 codePoint = static_cast<CodePoint>(firstByte & 0x07U);
234 } else {
235 ++current;
236 return static_cast<CodePoint>('?');
237 }
238
239 for (uint8_t index = 1; index < sequenceLength; ++index) {
240 const unsigned char nextByte = static_cast<unsigned char>(current[index]);
241 if (nextByte == 0 || (nextByte & 0xC0U) != 0x80U) {
242 ++current;
243 return static_cast<CodePoint>('?');
244 }
245
246 codePoint = static_cast<CodePoint>((codePoint << 6) | (nextByte & 0x3FU));
247 }
248
249 current += sequenceLength;
250 return codePoint;
251 }
