148 /**
149 * @brief Map an integer from one range to another.
150 *
151 * Unlike Arduino's map(), this function:
152 * - uses 64-bit intermediate arithmetic
153 * - explicitly handles invalid ranges
154 * - avoids accidental overflow for normal touch-controller ranges
155 */
156 static int32_t mapRange(int32_t value, int32_t inMin, int32_t inMax,
157 int32_t outMin, int32_t outMax) {
158 if (inMin == inMax) {
159 return outMin;
160 }
161
162 const int64_t numerator = static_cast<int64_t>(value - inMin) *
163 static_cast<int64_t>(outMax - outMin);
164
165 const int64_t denominator = static_cast<int64_t>(inMax - inMin);
166
167 return static_cast<int32_t>(outMin + numerator / denominator);
168 }

170 /**
171 * @brief Convert raw controller coordinates into logical display space.
172 *
173 * Processing order:
174 *
175 * raw coordinates
176 * -> inversion
177 * -> independent X/Y calibration
178 * -> swap XY
179 * -> display rotation
180 *
181 * This order is important. In particular, swapping raw X/Y before
182 * calibration would incorrectly apply the X calibration range to Y
183 * and vice versa.
184 */
185 Point mapCoordinates(int16_t rawX, int16_t rawY,
186 uint16_t pressure = 255) const {
187 if (!hasCalibration_) {
188 return applyRotation(rawX, rawY, pressure, false);
189 }
190
191 int32_t x = rawX;
192 int32_t y = rawY;
193
194 /*
195 * Inversion is performed in raw/controller space.
196 *
197 * Clamp first so a slightly out-of-range controller reading cannot
198 * produce surprising inverted values.
199 */
200 x = clamp32(x, calibration_.rawXMin, calibration_.rawXMax);
201
202 y = clamp32(y, calibration_.rawYMin, calibration_.rawYMax);
203
204 if (calibration_.invertX) {
205 x = calibration_.rawXMax - (x - calibration_.rawXMin);
206 }
207
208 if (calibration_.invertY) {
209 y = calibration_.rawYMax - (y - calibration_.rawYMin);
210 }
211
212 /*
213 * IMPORTANT:
214 *
215 * Map X against X calibration and Y against Y calibration BEFORE
216 * swapping axes.
217 */
218 int32_t mappedX =
219 mapRange(x, calibration_.rawXMin, calibration_.rawXMax, 0,
220 static_cast<int32_t>(calibration_.screenWidth) - 1);
221
222 int32_t mappedY =
223 mapRange(y, calibration_.rawYMin, calibration_.rawYMax, 0,
224 static_cast<int32_t>(calibration_.screenHeight) - 1);
225
226 mappedX =
227 clamp32(mappedX, 0, static_cast<int32_t>(calibration_.screenWidth) - 1);
228
229 mappedY = clamp32(mappedY, 0,
230 static_cast<int32_t>(calibration_.screenHeight) - 1);
231
232 /*
233 * Swap logical axes after calibration.
234 */
235 if (calibration_.swapXY) {
236 const int32_t tmp = mappedX;
237 mappedX = mappedY;
238 mappedY = tmp;
239 }
240
241 return applyRotation(mappedX, mappedY, pressure, true);
242 }
243
244 private:
245 Point applyRotation(int32_t x, int32_t y, uint16_t pressure,
246 bool calibrated) const {
247 if (!calibrated) {
248 return {static_cast<int16_t>(x), static_cast<int16_t>(y), pressure};
249 }
250
251 const int32_t width = static_cast<int32_t>(calibration_.screenWidth);
252
253 const int32_t height = static_cast<int32_t>(calibration_.screenHeight);
254
255 int32_t finalX = x;
256 int32_t finalY = y;
257
258 switch (rotation_) {
259 case Rotation::Deg0:
260 break;
261
262 case Rotation::Deg90:
263 /*
264 * Unrotated:
265 * X = [0, width-1]
266 * Y = [0, height-1]
267 *
268 * Rotated:
269 * X = [0, height-1]
270 * Y = [0, width-1]
271 */
272 finalX = height - 1 - y;
273 finalY = x;
274 break;
275
276 case Rotation::Deg180:
277 finalX = width - 1 - x;
278 finalY = height - 1 - y;
279 break;
280
281 case Rotation::Deg270:
282 finalX = y;
283 finalY = width - 1 - x;
284 break;
285 }
286
287 return {static_cast<int16_t>(finalX), static_cast<int16_t>(finalY),
288 pressure};
289 }

292/**
293 * @brief XPT2046 resistive touch controller.
294 *
295 * Uses SPI.
296 *
297 * XPT2046 command mapping:
298 * 0x90 = Y position
299 * 0xD0 = X position
300 * 0xB1 = Z1
301 * 0xC1 = Z2
302 */
303class TouchDriverXPT2046 : public TouchDriver {
304 public:
305 TouchDriverXPT2046(SPIClass& spi, int8_t csPin, int8_t irqPin = -1,
306 uint16_t zThreshold = 400)
307 : spi_(spi), csPin_(csPin), irqPin_(irqPin), zThreshold_(zThreshold) {}
308
309 bool begin() override {
310 if (csPin_ < 0) {
311 return false;
312 }
313
314 pinMode(csPin_, OUTPUT);
315 digitalWrite(csPin_, HIGH);
316
317 if (irqPin_ >= 0) {
318 pinMode(irqPin_, INPUT_PULLUP);
319 }
320
321 return true;
322 }
323
324 bool isTouched() override {
325 if (irqPin_ >= 0) {
326 return digitalRead(irqPin_) == LOW;
327 }
328
329 /*
330 * Without PENIRQ we need to perform an actual controller measurement.
331 */
332 Point point;
333 return getPoint(point);
334 }
335
336 bool getPoint(Point& outPoint) override {
337 spi_.beginTransaction(SPISettings(kSpiClock, MSBFIRST, SPI_MODE0));
338
339 digitalWrite(csPin_, LOW);
340
341 /*
342 * Read Z1 and Z2 first to determine whether the panel is pressed.
343 */
344 const uint16_t z1 = transfer12(0xB1);
345 const uint16_t z2 = transfer12(0xC1);
346
347 /*
348 * XPT2046's commonly used touch-presence metric:
349 *
350 * Z = Z1 + 4095 - Z2
351 *
352 * This is useful for thresholding but is NOT a calibrated physical
353 * pressure value in ohms or grams.
354 */
355 const int32_t z =
356 static_cast<int32_t>(z1) + 4095 - static_cast<int32_t>(z2);
357
358 if (z < static_cast<int32_t>(zThreshold_)) {
359 endTransaction();
360 return false;
361 }
362
363 /*
364 * IMPORTANT:
365 *
366 * 0xD0 = X
367 * 0x90 = Y
368 *
369 * The previous implementation had these reversed.
370 *
371 * Discard the first conversion for each axis. This improves stability
372 * on some XPT2046/panel combinations.
373 */
374 (void)transfer12(0xD0);
375 const uint16_t rawX = transfer12(0xD0);
376
377 (void)transfer12(0x90);
378 const uint16_t rawY = transfer12(0x90);
379
380 endTransaction();
381
382 /*
383 * Clamp Z to the representable 12-bit range before exposing it.
384 */
385 const uint16_t pressure = static_cast<uint16_t>(clamp32(z, 0, 4095));
386
387 outPoint = mapCoordinates(static_cast<int16_t>(rawX),
388 static_cast<int16_t>(rawY), pressure);
389
390 return true;
391 }
392
393 void setZThreshold(uint16_t threshold) { zThreshold_ = threshold; }
394
395 uint16_t getZThreshold() const { return zThreshold_; }
396
397 private:
398 static constexpr uint32_t kSpiClock = 2000000UL;
399
400 SPIClass& spi_;
401 int8_t csPin_;
402 int8_t irqPin_;
403 uint16_t zThreshold_;
404
405 uint16_t transfer12(uint8_t command) {
406 spi_.transfer(command);
407
408 const uint8_t high = spi_.transfer(0x00);
409
410 const uint8_t low = spi_.transfer(0x00);
411
412 return static_cast<uint16_t>(
413 ((static_cast<uint16_t>(high) << 8) | static_cast<uint16_t>(low)) >> 3);
414 }
415
416 void endTransaction() {
417 digitalWrite(csPin_, HIGH);
418 spi_.endTransaction();
419 }

422/**
423 * @brief FT6236 / FT6336 capacitive touch controller.
424 *
425 * I2C address is normally 0x38.
426 *
427 * Register layout used:
428 *
429 * 0x02 TD_STATUS
430 * 0x03 P1_XH
431 * 0x04 P1_XL
432 * 0x05 P1_YH
433 * 0x06 P1_YL
434 */
435class TouchDriverFT6236 : public TouchDriver {
436 public:
437 static constexpr uint8_t I2C_ADDR = 0x38;
438
439 TouchDriverFT6236(TwoWire& wire = Wire, int8_t rstPin = -1,
440 int8_t irqPin = -1)
441 : wire_(wire), rstPin_(rstPin), irqPin_(irqPin) {}
442
443 bool begin() override {
444 if (rstPin_ >= 0) {
445 pinMode(rstPin_, OUTPUT);
446
447 digitalWrite(rstPin_, LOW);
448 delay(10);
449
450 digitalWrite(rstPin_, HIGH);
451 delay(300);
452 }
453
454 if (irqPin_ >= 0) {
455 pinMode(irqPin_, INPUT_PULLUP);
456 }
457
458 /*
459 * The caller owns Wire.begin().
460 *
461 * We only verify that the device responds.
462 */
463 wire_.beginTransmission(I2C_ADDR);
464
465 return wire_.endTransmission() == 0;
466 }
467
468 bool isTouched() override {
469 if (irqPin_ >= 0 && digitalRead(irqPin_) == HIGH) {
470 return false;
471 }
472
473 uint8_t buffer[5];
474
475 if (!readRegisters(0x02, buffer, sizeof(buffer))) {
476 return false;
477 }
478
479 const uint8_t touches = buffer[0] & 0x0F;
480
481 return touches > 0;
482 }
483
484 bool getPoint(Point& outPoint) override {
485 if (irqPin_ >= 0 && digitalRead(irqPin_) == HIGH) {
486 return false;
487 }
488
489 /*
490 * Read status + P1 coordinates in one transaction:
491 *
492 * 0x02 TD_STATUS
493 * 0x03 P1_XH
494 * 0x04 P1_XL
495 * 0x05 P1_YH
496 * 0x06 P1_YL
497 */
498 uint8_t buffer[5];
499
500 if (!readRegisters(0x02, buffer, sizeof(buffer))) {
501 return false;
502 }
503
504 const uint8_t touches = buffer[0] & 0x0F;
505
506 if (touches == 0) {
507 return false;
508 }
509
510 const uint16_t rawX =
511 (static_cast<uint16_t>(buffer[1] & 0x0F) << 8) | buffer[2];
512
513 const uint16_t rawY =
514 (static_cast<uint16_t>(buffer[3] & 0x0F) << 8) | buffer[4];
515
516 /*
517 * Capacitive controller does not expose resistive pressure in the
518 * same sense as XPT2046. 255 means "touched".
519 */
520 outPoint = mapCoordinates(static_cast<int16_t>(rawX),
521 static_cast<int16_t>(rawY), 255);
522
523 return true;
524 }
525
526 private:
527 TwoWire& wire_;
528 int8_t rstPin_;
529 int8_t irqPin_;
530
531 bool readRegisters(uint8_t startRegister, uint8_t* buffer, size_t length) {
532 if (buffer == nullptr || length == 0) {
533 return false;
534 }
535
536 wire_.beginTransmission(I2C_ADDR);
537 wire_.write(startRegister);
538
539 if (wire_.endTransmission(false) != 0) {
540 return false;
541 }
542
543 const size_t received =
544 wire_.requestFrom(static_cast<int>(I2C_ADDR), static_cast<int>(length));
545
546 if (received != length) {
547 /*
548 * Drain anything the implementation may have left in the RX buffer.
549 */
550 while (wire_.available()) {
551 (void)wire_.read();
552 }
553
554 return false;
555 }
556
557 for (size_t i = 0; i < length; ++i) {
558 if (!wire_.available()) {
559 return false;
560 }
561
562 buffer[i] = static_cast<uint8_t>(wire_.read());
563 }
564
565 return true;
566 }
