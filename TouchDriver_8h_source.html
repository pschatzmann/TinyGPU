161 /**
162 * @brief Map an integer from one range to another.
163 *
164 * Unlike Arduino's map(), this function:
165 * - uses 64-bit intermediate arithmetic
166 * - explicitly handles invalid ranges
167 * - avoids accidental overflow for normal touch-controller ranges
168 */
169 static int32_t mapRange(int32_t value, int32_t inMin, int32_t inMax,
170 int32_t outMin, int32_t outMax) {
171 if (inMin == inMax) {
172 return outMin;
173 }
174
175 const int64_t numerator = static_cast<int64_t>(value - inMin) *
176 static_cast<int64_t>(outMax - outMin);
177
178 const int64_t denominator = static_cast<int64_t>(inMax - inMin);
179
180 return static_cast<int32_t>(outMin + numerator / denominator);
181 }

183 /**
184 * @brief Convert raw controller coordinates into logical display space.
185 *
186 * Processing order:
187 *
188 * raw coordinates
189 * -> inversion
190 * -> independent X/Y calibration
191 * -> swap XY
192 * -> display rotation
193 *
194 * This order is important. In particular, swapping raw X/Y before
195 * calibration would incorrectly apply the X calibration range to Y
196 * and vice versa.
197 */
198 Point mapCoordinates(int16_t rawX, int16_t rawY,
199 uint16_t pressure = 255) const {
200 if (!hasCalibration_) {
201 return applyRotation(rawX, rawY, pressure, false);
202 }
203
204 int32_t x = rawX;
205 int32_t y = rawY;
206
207 /*
208 * Inversion is performed in raw/controller space.
209 *
210 * Clamp first so a slightly out-of-range controller reading cannot
211 * produce surprising inverted values.
212 */
213 x = clamp32(x, calibration_.rawXMin, calibration_.rawXMax);
214
215 y = clamp32(y, calibration_.rawYMin, calibration_.rawYMax);
216
217 if (calibration_.invertX) {
218 x = calibration_.rawXMax - (x - calibration_.rawXMin);
219 }
220
221 if (calibration_.invertY) {
222 y = calibration_.rawYMax - (y - calibration_.rawYMin);
223 }
224
225 /*
226 * IMPORTANT:
227 *
228 * Map X against X calibration and Y against Y calibration BEFORE
229 * swapping axes.
230 */
231 int32_t mappedX =
232 mapRange(x, calibration_.rawXMin, calibration_.rawXMax, 0,
233 static_cast<int32_t>(calibration_.screenWidth) - 1);
234
235 int32_t mappedY =
236 mapRange(y, calibration_.rawYMin, calibration_.rawYMax, 0,
237 static_cast<int32_t>(calibration_.screenHeight) - 1);
238
239 mappedX =
240 clamp32(mappedX, 0, static_cast<int32_t>(calibration_.screenWidth) - 1);
241
242 mappedY = clamp32(mappedY, 0,
243 static_cast<int32_t>(calibration_.screenHeight) - 1);
244
245 /*
246 * Swap logical axes after calibration.
247 */
248 if (calibration_.swapXY) {
249 const int32_t tmp = mappedX;
250 mappedX = mappedY;
251 mappedY = tmp;
252 }
253
254 return applyRotation(mappedX, mappedY, pressure, true);
255 }
256
257 private:
258 Point applyRotation(int32_t x, int32_t y, uint16_t pressure,
259 bool calibrated) const {
260 if (!calibrated) {
261 return {static_cast<int16_t>(x), static_cast<int16_t>(y), pressure};
262 }
263
264 const int32_t width = static_cast<int32_t>(calibration_.screenWidth);
265
266 const int32_t height = static_cast<int32_t>(calibration_.screenHeight);
267
268 int32_t finalX = x;
269 int32_t finalY = y;
270
271 switch (rotation_) {
272 case Rotation::Deg0:
273 break;
274
275 case Rotation::Deg90:
276 /*
277 * Unrotated:
278 * X = [0, width-1]
279 * Y = [0, height-1]
280 *
281 * Rotated:
282 * X = [0, height-1]
283 * Y = [0, width-1]
284 */
285 finalX = height - 1 - y;
286 finalY = x;
287 break;
288
289 case Rotation::Deg180:
290 finalX = width - 1 - x;
291 finalY = height - 1 - y;
292 break;
293
294 case Rotation::Deg270:
295 finalX = y;
296 finalY = width - 1 - x;
297 break;
298 }
299
300 return {static_cast<int16_t>(finalX), static_cast<int16_t>(finalY),
301 pressure};
302 }

305/**
306 * @brief XPT2046 resistive touch controller.
307 *
308 * Uses SPI.
309 *
310 * XPT2046 command mapping:
311 * 0x90 = Y position
312 * 0xD0 = X position
313 * 0xB1 = Z1
314 * 0xC1 = Z2
315 */
316class TouchDriverXPT2046 : public TouchDriver {
317 public:
318 TouchDriverXPT2046(SPIClass& spi, int8_t csPin, int8_t irqPin = -1,
319 uint16_t zThreshold = 400)
320 : spi_(spi), csPin_(csPin), irqPin_(irqPin), zThreshold_(zThreshold) {}
321
322 bool begin() override {
323 if (csPin_ < 0) {
324 return false;
325 }
326
327 pinMode(csPin_, OUTPUT);
328 digitalWrite(csPin_, HIGH);
329
330 if (irqPin_ >= 0) {
331 pinMode(irqPin_, INPUT_PULLUP);
332 }
333
334 return true;
335 }
336
337 bool isTouched() override {
338 if (irqPin_ >= 0) {
339 return digitalRead(irqPin_) == LOW;
340 }
341
342 /*
343 * Without PENIRQ we need to perform an actual controller measurement.
344 */
345 Point point;
346 return getPoint(point);
347 }
348
349 bool getPoint(Point& outPoint) override {
350 spi_.beginTransaction(SPISettings(kSpiClock, MSBFIRST, SPI_MODE0));
351
352 digitalWrite(csPin_, LOW);
353
354 /*
355 * Read Z1 and Z2 first to determine whether the panel is pressed.
356 */
357 const uint16_t z1 = transfer12(0xB1);
358 const uint16_t z2 = transfer12(0xC1);
359
360 /*
361 * XPT2046's commonly used touch-presence metric:
362 *
363 * Z = Z1 + 4095 - Z2
364 *
365 * This is useful for thresholding but is NOT a calibrated physical
366 * pressure value in ohms or grams.
367 */
368 const int32_t z =
369 static_cast<int32_t>(z1) + 4095 - static_cast<int32_t>(z2);
370
371 if (z < static_cast<int32_t>(zThreshold_)) {
372 endTransaction();
373 return false;
374 }
375
376 /*
377 * IMPORTANT:
378 *
379 * 0xD0 = X
380 * 0x90 = Y
381 *
382 * The previous implementation had these reversed.
383 *
384 * Discard the first conversion for each axis. This improves stability
385 * on some XPT2046/panel combinations.
386 */
387 (void)transfer12(0xD0);
388 const uint16_t rawX = transfer12(0xD0);
389
390 (void)transfer12(0x90);
391 const uint16_t rawY = transfer12(0x90);
392
393 endTransaction();
394
395 /*
396 * Clamp Z to the representable 12-bit range before exposing it.
397 */
398 const uint16_t pressure = static_cast<uint16_t>(clamp32(z, 0, 4095));
399
400 outPoint = mapCoordinates(static_cast<int16_t>(rawX),
401 static_cast<int16_t>(rawY), pressure);
402
403 return true;
404 }
405
406 void setZThreshold(uint16_t threshold) { zThreshold_ = threshold; }
407
408 uint16_t getZThreshold() const { return zThreshold_; }
409
410 private:
411 static constexpr uint32_t kSpiClock = 2000000UL;
412
413 SPIClass& spi_;
414 int8_t csPin_;
415 int8_t irqPin_;
416 uint16_t zThreshold_;
417
418 uint16_t transfer12(uint8_t command) {
419 spi_.transfer(command);
420
421 const uint8_t high = spi_.transfer(0x00);
422
423 const uint8_t low = spi_.transfer(0x00);
424
425 return static_cast<uint16_t>(
426 ((static_cast<uint16_t>(high) << 8) | static_cast<uint16_t>(low)) >> 3);
427 }
428
429 void endTransaction() {
430 digitalWrite(csPin_, HIGH);
431 spi_.endTransaction();
432 }

456 bool begin() override {
457 if (rstPin_ >= 0) {
458 pinMode(rstPin_, OUTPUT);
459
460 digitalWrite(rstPin_, LOW);

464 delay(300);
465 }
466

469 }

484 }
485
486 uint8_t buffer[5];
487
488 if (!readRegisters(0x02, buffer, sizeof(buffer))) {
489 return false;

500 }

503 * Read status + P1 coordinates in one transaction:
504 *
505 * 0x02 TD_STATUS
506 * 0x03 P1_XH
507 * 0x04 P1_XL
508 * 0x05 P1_YH
509 * 0x06 P1_YL

512

514 return false;
515 }
516
517 const uint8_t touches = buffer[0] & 0x0F;
518
519 if (touches == 0) {
520 return false;
521 }
522

524 (static_cast<uint16_t>(buffer[1] & 0x0F) << 8) | buffer[2];
525

527 (static_cast<uint16_t>(buffer[3] & 0x0F) << 8) | buffer[4];
528

530 * Capacitive controller does not expose resistive pressure in the
531 * same sense as XPT2046. 255 means "touched".

534 static_cast<int16_t>(rawY), 255);

545 if (buffer == nullptr || length == 0) {
546 return false;
547 }
548
549 wire_.beginTransmission(I2C_ADDR);
550 wire_.write(startRegister);
551
552 if (wire_.endTransmission(false) != 0) {

558
559 if (received != length) {
560 /*

562 */
563 while (wire_.available()) {
564 (void)wire_.read();
565 }
566
567 return false;
568 }
569
570 for (size_t i = 0; i < length; ++i) {
571 if (!wire_.available()) {
572 return false;
573 }
574
575 buffer[i] = static_cast<uint8_t>(wire_.read());
576 }
577
578 return true;

730/**
731 * @brief GT911 capacitive touch controller.
732 *
733 * I2C address is normally 0x5D (alt 0x14, selected via the INT pin's level
734 * while RST is pulsed - see begin()). Unlike FT6236/CST816S, GT911's
735 * register addresses are 16-bit (MSB first), and its X/Y bytes are
736 * little-endian (low byte first) rather than the big-endian nibble-packed
737 * layout FT6236 uses.
738 *
739 * Register layout used:
740 *
741 * 0x8100 CONFIG_FRESH: write 1 after reset to make the chip start its
742 * sensing engine. Without this, the chip stays fully alive and
743 * correctly configured but never scans - status sits at 0x80
744 * ("config loaded, idle") forever.
745 * 0x80FF config checksum: round-tripped (read then written back
746 * unchanged) alongside the CONFIG_FRESH write.
747 * 0x814E status: bit7 = buffer ready, bits[3:0] = touch point count.
748 * Must be cleared (write 0x00) unconditionally on every poll,
749 * not only when a touch was found - clearing it only after a
750 * detected touch is a deadlock, since a stale, unacknowledged
751 * flag prevents the chip from ever reporting a new one.
752 * 0x814F point 0 data (7 bytes: track_id, x_lo, x_hi, y_lo, y_hi,
753 * size_lo, size_hi); point N starts at 0x814F + N*7.
754 *
755 * Because status must be cleared unconditionally exactly once per poll
756 * cycle, isTouched() does the chip's actual read/clear I/O and caches
757 * whatever point data was present; getPoint()/getSecondPoint() just
758 * return the cached result rather than re-querying the chip. Call
759 * isTouched() once per loop iteration (the same usage pattern as this
760 * file's other touch drivers) - calling it more than once between
761 * getPoint() calls will consume/clear a pending touch before getPoint()
762 * sees it.
763 *
764 * GT911 genuinely reports up to 5 simultaneous touches (unlike the
765 * single-touch FT6236/CST816S/XPT2046 parts also in this file), so
766 * getSecondPoint() is a real second contact, not a stub.
767 */
768class TouchDriverGT911 : public TouchDriver {
769 public:
770 static constexpr uint8_t I2C_ADDR_1 = 0x5D;
771 static constexpr uint8_t I2C_ADDR_2 = 0x14;
772
773 TouchDriverGT911(TwoWire& wire = Wire, int8_t rstPin = -1,
774 int8_t irqPin = -1, uint8_t i2cAddr = I2C_ADDR_1)
775 : wire_(wire), rstPin_(rstPin), irqPin_(irqPin), addr_(i2cAddr) {}
776
777 bool begin() override {
778 if (rstPin_ >= 0) {
779 /*
780 * Address-select reset sequence (GT911 datasheet): INT is held at
781 * the level matching the desired I2C address while RST is pulsed,
782 * then released to become the (unused, polled) touch IRQ line.
783 */
784 pinMode(rstPin_, OUTPUT);
785
786 if (irqPin_ >= 0) {
787 pinMode(irqPin_, OUTPUT);
788 digitalWrite(irqPin_, addr_ == I2C_ADDR_1 ? LOW : HIGH);
789 }
790
791 digitalWrite(rstPin_, LOW);
792 delay(10);
793
794 if (irqPin_ >= 0) {
795 digitalWrite(irqPin_, addr_ == I2C_ADDR_1 ? LOW : HIGH);
796 }
797 delay(1);
798
799 digitalWrite(rstPin_, HIGH);
800 delay(5);
801
802 if (irqPin_ >= 0) {
803 digitalWrite(irqPin_, LOW);
804 }
805 delay(50);
806 }
807
808 if (irqPin_ >= 0) {
809 pinMode(irqPin_, INPUT_PULLUP);
810 }
811
812 delay(50); // GT911 needs time after reset before it will ACK on I2C
813
814 wire_.beginTransmission(addr_);
815 if (wire_.endTransmission() != 0) {
816 return false;
817 }
818
819 /* Activate the sensing engine - see class comment. */
820 uint8_t checksum = 0;
821 readRegister(kRegConfigChecksum, checksum);
822 writeRegister(kRegConfigChecksum, checksum);
823 writeRegister(kRegConfigFresh, 0x01);
824 delay(10);
825
826 return true;
827 }
828
829 bool isTouched() override {
830 uint8_t status = 0;
831 readRegister(kRegStatus, status);
832 const bool bufferReady = (status & 0x80) != 0;
833 cachedTouches_ = status & 0x0F;
834
835 if (bufferReady && cachedTouches_ > 0) {
836 const uint8_t pointsToRead =
837 cachedTouches_ > 2 ? 2 : cachedTouches_; // this driver only caches 2
838
839 for (uint8_t i = 0; i < pointsToRead; ++i) {
840 uint8_t buf[7];
841 if (readRegisters(kRegPoint0 + i * 7, buf, sizeof(buf))) {
842 const uint16_t rawX = (static_cast<uint16_t>(buf[2]) << 8) | buf[1];
843 const uint16_t rawY = (static_cast<uint16_t>(buf[4]) << 8) | buf[3];
844 cachedPoints_[i] = mapCoordinates(static_cast<int16_t>(rawX),
845 static_cast<int16_t>(rawY), 255);
846 }
847 }
848 }
849
850 writeRegister(kRegStatus, 0x00); // unconditional - see class comment
851 return cachedTouches_ > 0;
852 }

854 bool getPoint(Point& outPoint) override {
855 if (cachedTouches_ == 0) return false;
856 outPoint = cachedPoints_[0];
857 return true;
858 }

860 bool getSecondPoint(Point& outPoint) override {
861 if (cachedTouches_ < 2) return false;
862 outPoint = cachedPoints_[1];
863 return true;
864 }

880 return readRegisters(reg, &value, 1);
881 }
882
883 bool readRegisters(uint16_t startRegister, uint8_t* buffer, size_t length) {
884 if (buffer == nullptr || length == 0) {
885 return false;
886 }
887
888 wire_.beginTransmission(addr_);
889 wire_.write(static_cast<uint8_t>(startRegister >> 8));
890 wire_.write(static_cast<uint8_t>(startRegister & 0xFF));
891
892 if (wire_.endTransmission(false) != 0) {
893 return false;
894 }
895
896 const size_t received =
897 wire_.requestFrom(static_cast<int>(addr_), static_cast<int>(length));
898
899 if (received != length) {
900 while (wire_.available()) {
901 (void)wire_.read();
902 }
903 return false;
904 }
905
906 for (size_t i = 0; i < length; ++i) {
907 if (!wire_.available()) {
908 return false;
909 }
910 buffer[i] = static_cast<uint8_t>(wire_.read());
911 }
912
913 return true;
914 }

916 void writeRegister(uint16_t reg, uint8_t value) {
917 wire_.beginTransmission(addr_);
918 wire_.write(static_cast<uint8_t>(reg >> 8));
919 wire_.write(static_cast<uint8_t>(reg & 0xFF));
920 wire_.write(value);
921 wire_.endTransmission();
922 }
