56 // Helper function to map raw coordinates to screen space based on calibration
57 // and rotation
58 Point mapCoordinates(int16_t rawX, int16_t rawY,
59 uint16_t pressure = 255) const {
60 if (!hasCalibration_) {
61 return {rawX, rawY, pressure};
62 }
63
64 int16_t x = rawX;
65 int16_t y = rawY;
66
67 // Apply raw inversion
68 if (calibration_.invertX)
69 x = calibration_.rawXMax - (x - calibration_.rawXMin);
70 if (calibration_.invertY)
71 y = calibration_.rawYMax - (y - calibration_.rawYMin);
72
73 // Swap axes if requested
74 if (calibration_.swapXY) {
75 int16_t tmp = x;
76 x = y;
77 y = tmp;
78 }
79
80 // Map raw values to screen pixel bounds
81 int32_t mappedX = map(x, calibration_.rawXMin, calibration_.rawXMax, 0,
82 calibration_.screenWidth);
83 int32_t mappedY = map(y, calibration_.rawYMin, calibration_.rawYMax, 0,
84 calibration_.screenHeight);
85
86 // Constrain
87 mappedX = constrain(mappedX, 0, (int32_t)calibration_.screenWidth - 1);
88 mappedY = constrain(mappedY, 0, (int32_t)calibration_.screenHeight - 1);
89
90 // Handle logical display rotation
91 int16_t finalX = mappedX;
92 int16_t finalY = mappedY;
93
94 switch (rotation_) {
95 case Rotation::Deg90:
96 finalX = calibration_.screenHeight - 1 - mappedY;
97 finalY = mappedX;
98 break;
99 case Rotation::Deg180:
100 finalX = calibration_.screenWidth - 1 - mappedX;
101 finalY = calibration_.screenHeight - 1 - mappedY;
102 break;
103 case Rotation::Deg270:
104 finalX = mappedY;
105 finalY = calibration_.screenWidth - 1 - mappedX;
106 break;
107 case Rotation::Deg0:
108 default:
109 break;
110 }
111
112 return {finalX, finalY, pressure};
113 }
114};

117 * @brief Driver for XPT2046 touch controller (resistive touch).
118 *
119 * Handles SPI communication and touch point reading for XPT2046.

121
122class TouchDriverXPT2046 : public TouchDriver {
123 public:
124 TouchDriverXPT2046(SPIClass& spi, int8_t csPin, int8_t irqPin = -1,
125 uint16_t zThreshold = 400)
126 : spi_(spi), csPin_(csPin), irqPin_(irqPin), zThreshold_(zThreshold) {}
127
128 bool begin() override {
129 pinMode(csPin_, OUTPUT);
130 digitalWrite(csPin_, HIGH);
131
132 if (irqPin_ >= 0) {
133 pinMode(irqPin_, INPUT_PULLUP);
134 }
135 return true;
136 }
137
138 bool isTouched() override {
139 if (irqPin_ >= 0) {
140 return digitalRead(irqPin_) == LOW;
141 }
142 Point dummy;
143 return getPoint(dummy);
144 }
145
146 bool getPoint(Point& outPoint) override {
147 spi_.beginTransaction(SPISettings(2000000, MSBFIRST, SPI_MODE0));
148 digitalWrite(csPin_, LOW);
149
150 // Read Z1 & Z2 to calculate pressure
151 uint16_t z1 = transfer12(0xB1);
152 uint16_t z2 = transfer12(0xC1);
153 uint16_t z = z1 + 4095 - z2;
154
155 if (z < zThreshold_) {
156 digitalWrite(csPin_, HIGH);
157 spi_.endTransaction();
158 return false;
159 }
160
161 // Read X & Y
162 transfer12(0x90); // Dummy read to settle
163 uint16_t rawX = transfer12(0x90);
164 uint16_t rawY = transfer12(0xD0);
165
166 digitalWrite(csPin_, HIGH);
167 spi_.endTransaction();
168
169 outPoint = mapCoordinates(rawX, rawY, z);
170 return true;
171 }
172
173 private:
174 SPIClass& spi_;
175 int8_t csPin_;
176 int8_t irqPin_;
177 uint16_t zThreshold_;
178
179 uint16_t transfer12(uint8_t cmd) {
180 spi_.transfer(cmd);
181 uint8_t high = spi_.transfer(0x00);
182 uint8_t low = spi_.transfer(0x00);
183 return ((high << 8) | low) >> 3;
184 }
185};

200 if (rstPin_ >= 0) {
201 pinMode(rstPin_, OUTPUT);
202 digitalWrite(rstPin_, LOW);

206 }
207
208 if (irqPin_ >= 0) {
209 pinMode(irqPin_, INPUT_PULLUP);
210 }
211
212 wire_.beginTransmission(I2C_ADDR);
213 return (wire_.endTransmission() == 0);
214 }
215
216 bool isTouched() override {
217 if (irqPin_ >= 0 && digitalRead(irqPin_) == HIGH) {
218 return false;
219 }
220 wire_.beginTransmission(I2C_ADDR);
221 wire_.write(0x02); // TD_STATUS register
222 if (wire_.endTransmission() != 0) return false;
223
224 wire_.requestFrom(I2C_ADDR, (uint8_t)1);
225 if (!wire_.available()) return false;
226
227 uint8_t touches = wire_.read() & 0x0F;
228 return (touches > 0 && touches <= 2);
229 }
230
231 bool getPoint(Point& outPoint) override {
232 if (!isTouched()) return false;
233
234 wire_.beginTransmission(I2C_ADDR);
235 wire_.write(0x03); // P1_XH register
236 if (wire_.endTransmission() != 0) return false;
237
238 wire_.requestFrom(I2C_ADDR, (uint8_t)4);
239 if (wire_.available() < 4) return false;
240
241 uint8_t xh = wire_.read();
242 uint8_t xl = wire_.read();
243 uint8_t yh = wire_.read();
244 uint8_t yl = wire_.read();
245
246 uint16_t rawX = ((xh & 0x0F) << 8) | xl;
247 uint16_t rawY = ((yh & 0x0F) << 8) | yl;
248
249 outPoint = mapCoordinates(rawX, rawY, 255);
250 return true;
251 }

253 private:

255 int8_t rstPin_;
256 int8_t irqPin_;
