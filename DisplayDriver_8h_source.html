16/// Common rotation values for rectangular display drivers (row/column
17/// exchange plus optional axis mirroring) - shared here, rather than
18/// duplicated as a nested enum per concrete driver (as ILI9341Driver/
19/// ILI9342Driver used to each define their own, identical, copy), so
20/// generic code holding only a DisplayDriver<RGB_T>& (e.g. an LCDBoard's
21/// display()) can call setRotation() without knowing the concrete driver
22/// type. Concrete drivers keep a `using Rotation = tinygpu::DisplayRotation;`
23/// alias so existing code written against e.g. ILI9341Driver<RGB565>::
24/// Rotation still compiles unchanged. Named DisplayRotation (not just
25/// Rotation) to avoid colliding with the differently-shaped touch
26/// tinygpu::Rotation (Deg0/Deg90/Deg180/Deg270) in TouchDriver.h.
27enum class DisplayRotation {
28 kNone = -1,
29 kPortrait = 0,
30 kLandscape = 1,
31 kPortraitFlipped = 2,
32 kLandscapeFlipped = 3,
33};
34
35/// True for the landscape family (kLandscape/kLandscapeFlipped) - false
36/// for portrait (kPortrait/kPortraitFlipped) or kNone. The landscape
37/// family has its row/column addressing exchanged relative to portrait
38/// (MV bit, on MIPI-DCS-style controllers), which is what makes
39/// width()/height() swap between the two families - a same-family
40/// rotation (e.g. kLandscape to kLandscapeFlipped, a 180 degree flip)
41/// only changes scan direction, not addressable width/height.
42inline bool isLandscapeFamily(DisplayRotation r) {
43 return r == DisplayRotation::kLandscape ||
44 r == DisplayRotation::kLandscapeFlipped;
45}
