34 bool writeData(ISurface<RGB_T>& surface, size_t x, size_t y) override {
35 static_assert(sizeof(RGB_T) == 2,
36 "writeData's byte-swap assumes a 16bpp RGB_T");
37 setAddressWindow(x, y, surface.width(), surface.height());
38 spi_.beginTransaction(SPISettings(frequencyHz_, MSBFIRST, SPI_MODE0));
39 digitalWrite(dc_, HIGH);
40 digitalWrite(cs_, LOW);
41 // RGB_T values are stored in the buffer in the host's native
42 // (little-endian) byte order, but the panel expects each pixel's
43 // high byte first on the wire - sending the buffer's raw memory
44 // order corrupts color across field boundaries (see writePixels's
45 // history in git log for the symptom this caused). Byte-swapped
46 // through a small scratch buffer and sent with the portable bulk
47 // SPIClass::transfer(void*, size_t) - available on every core, not
48 // just ESP32 - rather than one SPIClass::transfer16() call per
49 // pixel, since transfer() lets the underlying implementation batch
50 // many bytes into fewer hardware transactions. Can't byte-swap the
51 // surface's own buffer in place and bulk-transfer that directly:
52 // transfer(buf, count) is full-duplex and overwrites buf with
53 // received data, which would corrupt the framebuffer itself.
54 constexpr size_t kChunkBytes = 256;
55 uint8_t chunk[kChunkBytes];
56 const uint8_t* src = surface.data();
57 size_t remaining = surface.size();
58 while (remaining > 0) {
59 const size_t n = remaining < kChunkBytes ? remaining : kChunkBytes;
60 for (size_t i = 0; i < n; i += 2) {
61 chunk[i] = src[i + 1];
62 chunk[i + 1] = src[i];
63 }
64 spi_.transfer(chunk, n);
65 src += n;
66 remaining -= n;
67 }
68 digitalWrite(cs_, HIGH);
69 spi_.endTransaction();
70 return true;
71 }

73 bool readData(ISurface<RGB_T>& surface, size_t x, size_t y) {
74 size_t w = surface.width();
75 size_t h = surface.height();
76 size_t x0 = x + xOffset_;
77 size_t y0 = y + yOffset_;
78
79 spi_.beginTransaction(SPISettings(readFrequencyHz_, MSBFIRST, SPI_MODE0));
80 digitalWrite(cs_, LOW);
81
82 digitalWrite(dc_, LOW);
83 spi_.transfer(0x2A);
84 digitalWrite(dc_, HIGH);
85 spi_.transfer((x0) >> 8);
86 spi_.transfer((x0) & 0xFF);
87 spi_.transfer((x0 + w - 1) >> 8);
88 spi_.transfer((x0 + w - 1) & 0xFF);
89
90 digitalWrite(dc_, LOW);
91 spi_.transfer(0x2B);
92 digitalWrite(dc_, HIGH);
93 spi_.transfer((y0) >> 8);
94 spi_.transfer((y0) & 0xFF);
95 spi_.transfer((y0 + h - 1) >> 8);
96 spi_.transfer((y0 + h - 1) & 0xFF);
97
98 digitalWrite(dc_, LOW);
99 spi_.transfer(0x2E);
100 digitalWrite(dc_, HIGH);
101
102 spi_.transfer(0); // leading dummy byte before real pixel data
103 size_t pixelCount = w * h;
104 for (size_t i = 0; i < pixelCount; ++i) {
105 uint8_t r = spi_.transfer(0);
106 uint8_t g = spi_.transfer(0);
107 uint8_t b = spi_.transfer(0);
108 surface.setPixel(i % w, i / w, RGB_T(r, g, b));
109 }
110
111 digitalWrite(cs_, HIGH);
112 spi_.endTransaction();
113 return true;
114 }

357 /**

360 * this set, or every color renders as its photographic negative (RED
361 * shows as cyan, GREEN as magenta, BLUE as yellow) - confirmed on real

365

369 // MADCTL values with the BGR bit set (0x08), matching the MX/MY/MV
370 // (row/column/exchange) + BGR combinations Bodmer/TFT_eSPI's ILI9341

374 switch (rotation) {
375 case Rotation::kLandscape:

378 return 0x88; // MY | BGR
379 case Rotation::kLandscapeFlipped:
380 return 0xE8; // MX | MY | MV | BGR
381 case Rotation::kNone:
