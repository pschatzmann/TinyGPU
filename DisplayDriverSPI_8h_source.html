22 * panels. Lower it (e.g. 20000000 or 10000000) if you see glitches, which

46 /**
47 * @brief Writes a sub-region ("band") of pixel data starting at display
48 * coordinates (x, y).
49 *
50 * Useful on boards without enough contiguous RAM for a full-screen
51 * framebuffer: render a small band-sized surface and stream it to the
52 * display repeatedly at increasing y offsets instead of allocating one
53 * large buffer.
54 */
55 bool writeData(ISurface<RGB565>& surface, size_t x, size_t y) {
56 setAddressWindow(x, y, surface.width(), surface.height());
57 spi_.beginTransaction(SPISettings(frequencyHz_, MSBFIRST, SPI_MODE0));
58 digitalWrite(dc_, HIGH);
59 digitalWrite(cs_, LOW);
60 spi_.writeBytes(surface.data(), surface.size());
61 digitalWrite(cs_, HIGH);
62 spi_.endTransaction();
63 return true;
64 }

66 /**
67 * @brief Reads a rectangular region of pixels back from the display
68 * controller's own GRAM into `surface`, so a "sprite" background can be
69 * captured/restored without keeping a full local framebuffer.
70 *
71 * Uses the RAMRD (0x2E) command shared by most ILI9341/ST77xx-family
72 * controllers: after setting the column/row address window, the panel
73 * clocks out one dummy byte followed by 3 bytes per pixel (R, G, B, each
74 * with the color value in the top bits). This is read at a slower clock
75 * than writes (see readFrequencyHz in the constructor), since GRAM
76 * readback is weaker/less reliable at high speed on most of these parts.
77 *
78 * CS is held low continuously from the column/row address commands
79 * through the RAMRD command and every data byte - most of these
80 * controllers only return real GRAM data (rather than zeros/garbage) if
81 * that whole sequence happens in one unbroken chip-select period.
82 *
83 * @note Requires MISO to be wired to the panel. Some controller clones
84 * differ in their exact read timing/format or don't support reliable
85 * readback at all - verify this works correctly on your specific
86 * hardware before relying on it (e.g. read back a region you just wrote
87 * a known color to, and confirm the color matches).
88 */
89 bool readData(ISurface<RGB565>& surface, size_t x, size_t y) {
90 size_t w = surface.width();
91 size_t h = surface.height();
92 size_t x0 = x + xOffset_;
93 size_t y0 = y + yOffset_;
94
95 spi_.beginTransaction(SPISettings(readFrequencyHz_, MSBFIRST, SPI_MODE0));
96 digitalWrite(cs_, LOW);
97
98 digitalWrite(dc_, LOW);
99 spi_.transfer(0x2A);
100 digitalWrite(dc_, HIGH);
101 spi_.transfer((x0) >> 8);
102 spi_.transfer((x0) & 0xFF);
103 spi_.transfer((x0 + w - 1) >> 8);
104 spi_.transfer((x0 + w - 1) & 0xFF);
105
106 digitalWrite(dc_, LOW);
107 spi_.transfer(0x2B);
108 digitalWrite(dc_, HIGH);
109 spi_.transfer((y0) >> 8);
110 spi_.transfer((y0) & 0xFF);
111 spi_.transfer((y0 + h - 1) >> 8);
112 spi_.transfer((y0 + h - 1) & 0xFF);
113
114 digitalWrite(dc_, LOW);
115 spi_.transfer(0x2E);
116 digitalWrite(dc_, HIGH);
117
118 spi_.transfer(0); // leading dummy byte before real pixel data
119 size_t pixelCount = w * h;
120 for (size_t i = 0; i < pixelCount; ++i) {
121 uint8_t r = spi_.transfer(0);
122 uint8_t g = spi_.transfer(0);
123 uint8_t b = spi_.transfer(0);
124 surface.setPixel(i % w, i / w, RGB565(r, g, b));
125 }
126
127 digitalWrite(cs_, HIGH);
128 spi_.endTransaction();
129 return true;
130 }

240/**
241 * @brief Driver for ILI9341 SPI display controller.
242 *
243 * Handles initialization and address window logic for ILI9341 displays.
244 */
245class ILI9341Driver : public DisplayDriverSPI {
246 public:
247 /**
248 * @brief Display orientation, sent via the MADCTL command (0x36) -
249 * same convention as Adafruit_ILI9341::setRotation(). kNone means
250 * "don't send MADCTL at all" - the panel's power-on-reset default
251 * orientation, preserving this driver's original (pre-rotation-support)
252 * behavior exactly for existing code that doesn't request a rotation.
253 */
254 enum class Rotation {
255 kNone = -1,
256 kPortrait = 0,
257 kLandscape = 1,
258 kPortraitFlipped = 2,
259 kLandscapeFlipped = 3,
260 };
261
262 ILI9341Driver(SPIClass& spi, int8_t cs, int8_t dc, int8_t rst = -1,
263 Rotation rotation = Rotation::kNone)
264 : DisplayDriverSPI(spi, cs, dc, rst, 0, 0), rotation_(rotation) {}
265
266 bool begin() override {
267 setupPinsAndReset();
268 writeCommand(0x01);
269 delay(150);
270 writeCommand(0x28);
271 writeCommand(0x3A);
272 writeData8(0x55);
273 if (rotation_ != Rotation::kNone) {
274 writeCommand(0x36);
275 writeData8(madctlForRotation(rotation_));
276 }
277 writeCommand(0x11);
278 delay(120);
279 writeCommand(0x29);
280 return true;
281 }
282
283 /**
284 * @brief Changes the display orientation after begin() - sends MADCTL
285 * (0x36) directly, so it takes effect immediately without a full
286 * re-init. Rotation::kNone is ignored.
287 */
288 void setRotation(Rotation rotation) {
289 if (rotation == Rotation::kNone) return;
290 rotation_ = rotation;
291 writeCommand(0x36);
292 writeData8(madctlForRotation(rotation_));
293 }
294
295 /// The rotation passed to the constructor or setRotation(), or
296 /// Rotation::kNone if neither has been called (MADCTL was never sent).
297 Rotation rotation() const { return rotation_; }
298
299 protected:
300 Rotation rotation_;
301
302 /// Adafruit_ILI9341-compatible MADCTL byte for `rotation`.
303 static uint8_t madctlForRotation(Rotation rotation) {
304 switch (rotation) {
305 case Rotation::kPortrait:
306 return 0x48; // portrait: MX | BGR
307 case Rotation::kLandscape:
308 return 0x28; // landscape: MV | BGR
309 case Rotation::kPortraitFlipped:
310 return 0x88; // portrait, flipped 180: MY | BGR
311 default:
312 return 0xE8; // landscape, flipped 180: MX | MY | MV | BGR
313 }
314 }
315};
