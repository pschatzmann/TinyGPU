34 /// Swaps each pixel's two bytes right before they go out over SPI (see
35 /// writeData()) - off by default, since RGB_T (RGB565) already stores
36 /// values pre-swapped to the wire order most panels expect (see
37 /// RGB565.h). Some panels/wiring combinations expect the opposite of
38 /// that default (symptom: colors come out scrambled/inverted-looking -
39 /// e.g. a purple app bar rendering green, a light background rendering
40 /// black - across *everything* drawn, not just images); this is the one
41 /// place all pixel data funnels through before hitting the wire, so
42 /// flipping it here fixes fills/text/images uniformly instead of needing
43 /// a per-source fix (contrast JPEGParser::setSwapBytes(), which only
44 /// affects decoded images).
45 void setSwapOutputBytes(bool swap) { swapOutputBytes_ = swap; }

47 bool writeData(ISurface<RGB_T>& surface, size_t x, size_t y) override {
48 static_assert(sizeof(RGB_T) == 2,
49 "writeData assumes a 16bpp RGB_T (RGB565) stored in "
50 "wire byte order");
51 setAddressWindow(x, y, surface.width(), surface.height());
52 spi_.beginTransaction(SPISettings(frequencyHz_, MSBFIRST, SPI_MODE0));
53 digitalWrite(dc_, HIGH);
54 digitalWrite(cs_, LOW);
55 // RGB_T (RGB565) values are stored in the buffer already byte-swapped
56 // to the panel's wire order (see RGB565.h), so the buffer's raw memory
57 // order can normally be sent as-is - no per-pixel swap needed here.
58 // swapOutputBytes_ (see setSwapOutputBytes()) undoes that for panels
59 // that need the opposite order.
60 // Still copied through a small scratch buffer and sent with the
61 // portable bulk SPIClass::transfer(void*, size_t) - available on every
62 // core, not just ESP32 - rather than one SPIClass::transfer16() call
63 // per pixel, since transfer() lets the underlying implementation batch
64 // many bytes into fewer hardware transactions. Can't bulk-transfer the
65 // surface's own buffer directly: transfer(buf, count) is full-duplex
66 // and overwrites buf with received data, which would corrupt the
67 // framebuffer itself.
68 constexpr size_t kChunkBytes = 256;
69 uint8_t chunk[kChunkBytes];
70 const uint8_t* src = surface.data();
71 size_t remaining = surface.size();
72 while (remaining > 0) {
73 const size_t n = remaining < kChunkBytes ? remaining : kChunkBytes;
74 memcpy(chunk, src, n);
75 if (swapOutputBytes_) {
76 // n is always even here: surface.size() is width*height*sizeof(RGB_T)
77 // with sizeof(RGB_T) == 2 (see the static_assert above), and
78 // kChunkBytes is even, so every chunk - including the last,
79 // possibly-partial one - covers whole pixels.
80 for (size_t i = 0; i + 1 < n; i += 2) {
81 const uint8_t tmp = chunk[i];
82 chunk[i] = chunk[i + 1];
83 chunk[i + 1] = tmp;
84 }
85 }
86 spi_.transfer(chunk, n);
87 src += n;
88 remaining -= n;
89 }
90 digitalWrite(cs_, HIGH);
91 spi_.endTransaction();
92 return true;
93 }

95 bool readData(ISurface<RGB_T>& surface, size_t x, size_t y) {
96 size_t w = surface.width();
97 size_t h = surface.height();
98 size_t x0 = x + xOffset_;
99 size_t y0 = y + yOffset_;
100
101 spi_.beginTransaction(SPISettings(readFrequencyHz_, MSBFIRST, SPI_MODE0));
102 digitalWrite(cs_, LOW);
103
104 digitalWrite(dc_, LOW);
105 spi_.transfer(0x2A);
106 digitalWrite(dc_, HIGH);
107 spi_.transfer((x0) >> 8);
108 spi_.transfer((x0) & 0xFF);
109 spi_.transfer((x0 + w - 1) >> 8);
110 spi_.transfer((x0 + w - 1) & 0xFF);
111
112 digitalWrite(dc_, LOW);
113 spi_.transfer(0x2B);
114 digitalWrite(dc_, HIGH);
115 spi_.transfer((y0) >> 8);
116 spi_.transfer((y0) & 0xFF);
117 spi_.transfer((y0 + h - 1) >> 8);
118 spi_.transfer((y0 + h - 1) & 0xFF);
119
120 digitalWrite(dc_, LOW);
121 spi_.transfer(0x2E);
122 digitalWrite(dc_, HIGH);
123
124 spi_.transfer(0); // leading dummy byte before real pixel data
125 size_t pixelCount = w * h;
126 for (size_t i = 0; i < pixelCount; ++i) {
127 uint8_t r = spi_.transfer(0);
128 uint8_t g = spi_.transfer(0);
129 uint8_t b = spi_.transfer(0);
130 surface.setPixel(i % w, i / w, RGB_T(r, g, b));
131 }
132
133 digitalWrite(cs_, HIGH);
134 spi_.endTransaction();
135 return true;
136 }

400 /**
401 * @brief Toggles the ILI9341's display inversion (DINVON/DINVOFF, 0x21/

403 * this set, or every color renders as its photographic negative (RED
404 * shows as cyan, GREEN as magenta, BLUE as yellow) - confirmed on real
405 * hardware. Safe to call any time after begin() - and safe to call
406 * *before* a later begin() too: begin() sends SWRESET (0x01), a real
407 * hardware reset that puts the panel's own inversion register back to
408 * its power-on default regardless of what this method last told it (no
409 * explicit invert command appears anywhere in begin() itself, so this

413 * caller's own redundant DeviceOutput::begin()).

416 invertColor_ = invert;
417 writeCommand(invert ? 0x21 : 0x20);
418 }

436 case Rotation::kLandscapeFlipped:
437 return 0xE8; // MX | MY | MV | BGR
438 case Rotation::kNone:
439 case Rotation::kPortrait:

442 }
