44 bool readData(ISurface<RGB_T>& surface, size_t x, size_t y) {
45 size_t w = surface.width();
46 size_t h = surface.height();
47 size_t x0 = x + xOffset_;
48 size_t y0 = y + yOffset_;
49
50 spi_.beginTransaction(SPISettings(readFrequencyHz_, MSBFIRST, SPI_MODE0));
51 digitalWrite(cs_, LOW);
52
53 digitalWrite(dc_, LOW);
54 spi_.transfer(0x2A);
55 digitalWrite(dc_, HIGH);
56 spi_.transfer((x0) >> 8);
57 spi_.transfer((x0) & 0xFF);
58 spi_.transfer((x0 + w - 1) >> 8);
59 spi_.transfer((x0 + w - 1) & 0xFF);
60
61 digitalWrite(dc_, LOW);
62 spi_.transfer(0x2B);
63 digitalWrite(dc_, HIGH);
64 spi_.transfer((y0) >> 8);
65 spi_.transfer((y0) & 0xFF);
66 spi_.transfer((y0 + h - 1) >> 8);
67 spi_.transfer((y0 + h - 1) & 0xFF);
68
69 digitalWrite(dc_, LOW);
70 spi_.transfer(0x2E);
71 digitalWrite(dc_, HIGH);
72
73 spi_.transfer(0); // leading dummy byte before real pixel data
74 size_t pixelCount = w * h;
75 for (size_t i = 0; i < pixelCount; ++i) {
76 uint8_t r = spi_.transfer(0);
77 uint8_t g = spi_.transfer(0);
78 uint8_t b = spi_.transfer(0);
79 surface.setPixel(i % w, i / w, RGB_T(r, g, b));
80 }
81
82 digitalWrite(cs_, HIGH);
83 spi_.endTransaction();
84 return true;
85 }
