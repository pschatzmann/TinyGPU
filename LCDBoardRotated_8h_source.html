1// TinyGPU - 90 degree rotation of any LCDBoard.
2//
3// A board with a landscape panel can show the UI in portrait (or the other way
4// round). LCDBoardRotated wraps a board and reports the rotated size. The image
5// is rotated in hardware if the display driver supports setRotation() (its
6// width() and height() change); otherwise it is rotated in software, one row at a
7// time, so no frame buffer is needed. Touch points are mapped to the same
8// rotated coordinates.
9//
10// LCDBoardDesktopSDL board(320, 240); // landscape panel
11// LCDBoardRotated rotated(board); // 240 x 320 portrait UI
12// tinymd::App<RGB565> app(rotated, theme);
13//
14// Rotations (the UI is portrait):
15// kPortrait - clockwise: the top left of the UI is at the top right of the panel
16// kPortraitFlipped - counter-clockwise: the top left of the UI is at the bottom left

31// Maps a portrait coordinate to the landscape panel (rotation: kPortrait or kPortraitFlipped).
32// portraitW/H are the rotated (UI) size; the panel is portraitH x portraitW.
33inline void rotatePoint(DisplayRotation rotation, size_t portraitW, size_t portraitH, size_t px, size_t py,
34 size_t& lx, size_t& ly) {
35 if (rotation == DisplayRotation::kPortraitFlipped) {
36 lx = py;
37 ly = portraitW - 1 - px;
38 } else {
39 lx = portraitH - 1 - py;
40 ly = px;
41 }
42}
43
44// The inverse of rotatePoint(): panel coordinates to the UI.
45inline void unrotatePoint(DisplayRotation rotation, size_t portraitW, size_t portraitH, size_t lx, size_t ly,
46 size_t& px, size_t& py) {
47 if (rotation == DisplayRotation::kPortraitFlipped) {
48 px = portraitW - 1 - ly;
49 py = lx;
50 } else {
51 px = ly;
52 py = portraitH - 1 - lx;
53 }
54}

58/// A display driver that shows its UI rotated by 90 degrees. Hardware rotation is
59/// used when the driver supports it (see LCDBoardRotated); otherwise the rows are
60/// rotated in software when they are written.
61class RotatedDisplay : public DisplayDriver<RGB565> {
62 public:
63 RotatedDisplay(DisplayDriver<RGB565>& panel, size_t portraitW, size_t portraitH, DisplayRotation rotation)
64 : panel_(panel), portraitW_(portraitW), portraitH_(portraitH), rotation_(rotation) {}
65
66 /// The board begins the panel; this only exists for the App, which calls it too.
67 bool begin() override { return true; }
68
69 bool writeData(ISurface<RGB565>& surface) override { return writeData(surface, 0, 0); }
70
71 /// Writes the UI region at (x, y) (UI coordinates).
72 bool writeData(ISurface<RGB565>& surface, size_t x, size_t y) override {
73 if (hardware_) return panel_.writeData(surface, x, y);
74 const size_t bw = surface.width(), bh = surface.height();
75 if (bw == 0 || bh == 0) return true;
76 // a row of the panel is one column of the UI region
77 Surface<RGB565> row(bh, 1, defaultFont);
78 if (!row.begin()) return false;
79 if (rotation_ == DisplayRotation::kPortraitFlipped) {
80 // panel column y+k is the UI row k of the region, read from the right
81 const size_t ly0 = portraitW_ - (x + bw), lx0 = y;
82 for (size_t r = 0; r < bw; ++r) {
83 const size_t sx = bw - 1 - r;
84 for (size_t k = 0; k < bh; ++k) row.setPixel(k, 0, surface.getPixel(sx, k));
85 if (!panel_.writeData(row, lx0, ly0 + r)) return false;
86 }
87 } else {
88 // the UI column x+r runs down the panel row x+r, from the bottom up
89 const size_t ly0 = x, lx0 = portraitH_ - (y + bh);
90 for (size_t r = 0; r < bw; ++r) {
91 for (size_t k = 0; k < bh; ++k) row.setPixel(k, 0, surface.getPixel(r, bh - 1 - k));
92 if (!panel_.writeData(row, lx0, ly0 + r)) return false;
93 }
94 }
95 return true;
96 }
97
98 /// The UI size (the rotated size).
99 size_t width() const override { return portraitW_; }
100 size_t height() const override { return portraitH_; }
101
102 /// True if the panel rotates the image itself.
103 bool hardwareRotation() const { return hardware_; }
104 void setHardwareRotation(bool hardware) { hardware_ = hardware; }
105 DisplayRotation rotation() const { return rotation_; }
106
107 protected:
108 bool setAddressWindow(size_t, size_t, size_t, size_t) override { return true; }
109
110 private:
111 DisplayDriver<RGB565>& panel_;
112 size_t portraitW_;
113 size_t portraitH_;
114 DisplayRotation rotation_;
115 bool hardware_ = false;
116};

118/// A touch controller whose points are in the rotated (UI) coordinates.
119class RotatedTouch : public TouchDriver {
120 public:
121 RotatedTouch(TouchDriver& touch, size_t portraitW, size_t portraitH, DisplayRotation rotation)
122 : touch_(touch), portraitW_(portraitW), portraitH_(portraitH), rotation_(rotation) {}
123
124 bool begin() override { return true; }
125 bool isTouched() override { return touch_.isTouched(); }
126
127 bool getPoint(Point& outPoint) override {
128 Point p;
129 if (!touch_.getPoint(p)) return false;
130 size_t px = 0, py = 0;
131 detail::unrotatePoint(rotation_, portraitW_, portraitH_, static_cast<size_t>(p.x), static_cast<size_t>(p.y), px,
132 py);
133 outPoint = p;
134 outPoint.x = static_cast<int16_t>(px);
135 outPoint.y = static_cast<int16_t>(py);
136 return true;
137 }
138
139 private:
140 TouchDriver& touch_;
141 size_t portraitW_;
142 size_t portraitH_;
143 DisplayRotation rotation_;
144};

146/// Wraps a board: its width() and height() are the rotated size when the panel is
147/// landscape. The rotation is applied in hardware when the display driver supports
148/// it, otherwise in software.
149class LCDBoardRotated : public LCDBoard {
150 public:
151 /// Wraps `board`, which must outlive this object. A panel that is already portrait
152 /// is not rotated.
153 explicit LCDBoardRotated(LCDBoard& board, DisplayRotation rotation = DisplayRotation::kPortrait)
154 : board_(board),
155 rotated_(board.width() > board.height()),
156 display_(board.display(), rotated_ ? board.height() : board.width(), rotated_ ? board.width() : board.height(),
157 rotation),
158 touch_(board.touch() != nullptr
159 ? new RotatedTouch(*board.touch(), display_.width(), display_.height(), rotation)
160 : nullptr) {}
161
162 ~LCDBoardRotated() override { delete touch_; }
163
164 /// Begins the board, then asks the panel to rotate itself. If the panel's size does
165 /// not change, the rotation is done in software.
166 bool begin() override {
167 if (!board_.begin()) return false;
168 if (rotated_) {
169 DisplayDriver<RGB565>& panel = board_.display();
170 panel.setRotation(display_.rotation());
171 display_.setHardwareRotation(panel.width() == display_.width() && panel.height() == display_.height());
172 }
173 return true;
174 }
175
176 size_t width() const override { return rotated_ ? display_.width() : board_.width(); }
177 size_t height() const override { return rotated_ ? display_.height() : board_.height(); }
178
179 /// The rotated display, or the board's own display if the panel is portrait.
180 DisplayDriver<RGB565>& display() override { return rotated_ ? static_cast<DisplayDriver<RGB565>&>(display_) : board_.display(); }
181 TouchDriver* touch() override { return rotated_ && touch_ != nullptr ? touch_ : board_.touch(); }
182 const I2SPins& i2s() const override { return board_.i2s(); }
183 const LEDPins& led() const override { return board_.led(); }
184 int8_t backlightPin() const override { return board_.backlightPin(); }
185
186 /// True if the board was rotated (the panel is landscape).
187 bool rotated() const { return rotated_; }
188 /// True if the panel rotates the image itself (valid after begin()).
189 bool hardwareRotation() const { return display_.hardwareRotation(); }
190
191 private:
192 LCDBoard& board_;
193 bool rotated_;
194 RotatedDisplay display_;
195 RotatedTouch* touch_;
196};
