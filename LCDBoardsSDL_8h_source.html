18/**
19 * @brief Desktop SDL2 window standing in for a physical LCD board. Window
20 * size is set at construction time (default 320x240) since - unlike the
21 * hardware boards in LCDBoardsESP32.h - there is no fixed panel to match.
22 *
23 * The left mouse button stands in for a finger (see TouchDriverSDL).
24 * There is no backlight/I2S/LED hardware on this backend, so i2s()/led()
25 * report "not present" (every field -1) and backlightPin() is -1.
26 */
27class LCDBoardDesktopSDL : public LCDBoard {
28 public:
29 LCDBoardDesktopSDL(size_t width = 320, size_t height = 240)

32 /// Opens the SDL window and starts mouse-as-touch tracking.
33 bool begin() override {

37

39 size_t width() const override { return width_; }

42
43 /// The board's display driver.

45 /// The board's touch controller (mouse-mapped).
46 TouchDriver* touch() override { return &touch_; }
47 /// No I2S bus on this backend - every field is -1.
48 const I2SPins& i2s() const override { return i2s_; }
49 /// No LED on this backend - every field is -1.
50 const LEDPins& led() const override { return led_; }
51
52 private:
