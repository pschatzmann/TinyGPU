9/// Lifecycle phase of a continuous gesture (drag/pan/scroll/pinch/rotate).
10/// Discrete gestures (tap/double_tap/long_press/swipe_*) always report
11/// kEnded, since they are recognized once, at the moment they complete.
12enum class GesturePhase : uint8_t { kBegan, kChanged, kEnded };
13
14enum class GestureType : uint8_t {
15 kNone,
16 kTap,

19 kSwipeLeft,

24 kPan, ///< continuous move over the background, not vertical-dominant
25 kScroll, ///< continuous move over the background, vertical-dominant

79/**
80 * @brief Recognizes touch gestures from a stream of single-point touch
81 * samples, and reports them through a single callback.
82 *
83 * This does not replace TouchDriver - feed it the same driver you'd
84 * otherwise poll directly, once per loop():
85 *
86 * GestureDetector gestures;
87 * gestures.onGesture = handleGesture;
88 * gestures.isDraggable = [](int16_t x, int16_t y) { ... }; // optional
89 * void loop() { gestures.update(touchDriver); }
90 *
91 * isDraggable, if set, is checked against the position where a touch
92 * started; if it returns true the resulting continuous-move gesture is
93 * reported as kDrag, otherwise as kPan/kScroll. Without it, continuous
94 * moves are always reported as kPan/kScroll (never kDrag).
95 *
96 * pinch/rotate require TouchDriver::getSecondPoint() to return a genuine
97 * second simultaneous touch; none of the drivers built into this library
98 * can supply one (see TouchDriver::getSecondPoint), so those two gesture
99 * types will not fire against them.
100 */
101class GestureDetector {
102 public:
103 void (*onGesture)(GestureEvent&) = nullptr;
104 bool (*isDraggable)(int16_t x, int16_t y) = nullptr;
105
106 // Tuning knobs, in pixels / milliseconds.
107 uint16_t tapMaxDurationMs = 300;
108 uint16_t tapMaxMovePx = 12;
109 uint16_t doubleTapMaxGapMs = 350;
110 uint16_t longPressMinDurationMs = 600;
111 uint16_t swipeMinDistancePx = 40;
112 uint16_t swipeMaxDurationMs = 500;
113 uint16_t dragStartThresholdPx = 8;
114
115 /// Call once per loop() with the same driver you'd otherwise poll
116 /// directly via isTouched()/getPoint().
117 void update(TouchDriver& driver) {
118 Point point;
119 const bool touched = driver.isTouched() && driver.getPoint(point);
120 const uint32_t now = millis();
121
122 if (touched) {
123 handleTouched(point, now);
124 Point second;
125 updatePinchRotate(driver.getSecondPoint(second), point, second);
126 } else if (isDown_) {
127 handleReleased(now);
128 }
129 }
130
131 private:
132 bool isDown_ = false;
133 bool dragActive_ = false;
134 bool longPressFired_ = false;
135 bool lastTapValid_ = false;
136 GestureType dragType_ = GestureType::kPan;
137 uint32_t downTime_ = 0;
138 uint32_t lastTapTime_ = 0;
139 Point downPoint_;
140 Point lastPoint_;
141
142 bool secondActive_ = false;
143 float initialPinchDistance_ = 1.0f;
144 float lastPinchDistance_ = 0.0f;
145 float initialPinchAngle_ = 0.0f;
146 float lastPinchAngle_ = 0.0f;
147
148 static int16_t iabs(int16_t v) { return v < 0 ? static_cast<int16_t>(-v) : v; }
149
150 void handleTouched(Point point, uint32_t now) {
151 if (!isDown_) {
152 isDown_ = true;
153 downTime_ = now;
154 downPoint_ = point;
155 lastPoint_ = point;
156 dragActive_ = false;
157 longPressFired_ = false;
158 return;
159 }
160
161 const int16_t dx = static_cast<int16_t>(point.x - downPoint_.x);
162 const int16_t dy = static_cast<int16_t>(point.y - downPoint_.y);
163 const int16_t adx = iabs(dx);
164 const int16_t ady = iabs(dy);
165
166 if (!dragActive_ && (adx > dragStartThresholdPx || ady > dragStartThresholdPx)) {
167 dragActive_ = true;
168 dragType_ = classifyDragType(downPoint_, dx, dy);
169 emitContinuous(dragType_, GesturePhase::kBegan, point, dx, dy, 0, 0, now);
170 } else if (dragActive_) {
171 const int16_t stepDx = static_cast<int16_t>(point.x - lastPoint_.x);
172 const int16_t stepDy = static_cast<int16_t>(point.y - lastPoint_.y);
173 if (stepDx != 0 || stepDy != 0) {
174 emitContinuous(dragType_, GesturePhase::kChanged, point, dx, dy, stepDx, stepDy, now);
175 }
176 }
177
178 if (!dragActive_ && !longPressFired_ && (now - downTime_) >= longPressMinDurationMs) {
179 longPressFired_ = true;
180 emitDiscrete(GestureType::kLongPress, point, dx, dy, now - downTime_);
181 }
182
183 lastPoint_ = point;
184 }
185
186 void handleReleased(uint32_t now) {
187 isDown_ = false;
188 const uint32_t duration = now - downTime_;
189 const int16_t dx = static_cast<int16_t>(lastPoint_.x - downPoint_.x);
190 const int16_t dy = static_cast<int16_t>(lastPoint_.y - downPoint_.y);
191 const int16_t adx = iabs(dx);
192 const int16_t ady = iabs(dy);
193
194 if (dragActive_) {
195 emitContinuous(dragType_, GesturePhase::kEnded, lastPoint_, dx, dy, 0, 0, now);
196 // A drag/pan/scroll that was also fast and far enough is additionally
197 // reported as a swipe, so callers that only care about swipes don't
198 // need to track distance/duration themselves.
199 if (duration <= swipeMaxDurationMs &&
200 (adx >= swipeMinDistancePx || ady >= swipeMinDistancePx)) {
201 emitDiscrete(swipeTypeFor(dx, dy, adx, ady), lastPoint_, dx, dy, duration);
202 }
203 dragActive_ = false;
204 } else if (!longPressFired_ && adx <= tapMaxMovePx && ady <= tapMaxMovePx &&
205 duration <= tapMaxDurationMs) {
206 if (lastTapValid_ && (downTime_ - lastTapTime_) <= doubleTapMaxGapMs) {
207 emitDiscrete(GestureType::kDoubleTap, lastPoint_, dx, dy, duration);
208 lastTapValid_ = false;
209 } else {
210 emitDiscrete(GestureType::kTap, lastPoint_, dx, dy, duration);
211 lastTapValid_ = true;
212 lastTapTime_ = downTime_;
213 }
214 }
215
216 secondActive_ = false;
217 }
218
219 GestureType classifyDragType(Point downPoint, int16_t dx, int16_t dy) const {
220 if (isDraggable != nullptr && isDraggable(downPoint.x, downPoint.y)) {
221 return GestureType::kDrag;
222 }
223 return (iabs(dy) >= iabs(dx)) ? GestureType::kScroll : GestureType::kPan;
224 }
225
226 static GestureType swipeTypeFor(int16_t dx, int16_t dy, int16_t adx, int16_t ady) {
227 if (adx >= ady) {
228 return dx > 0 ? GestureType::kSwipeRight : GestureType::kSwipeLeft;
229 }
230 return dy > 0 ? GestureType::kSwipeDown : GestureType::kSwipeUp;
231 }

233 void emitContinuous(GestureType type, GesturePhase phase, Point point, int16_t dx,
234 int16_t dy, int16_t stepDx, int16_t stepDy, uint32_t now) {
235 if (!onGesture) return;
236 GestureEvent ev;
237 ev.type = type;
238 ev.phase = phase;
239 ev.point = point;
240 ev.startPoint = downPoint_;
241 ev.deltaX = dx;
242 ev.deltaY = dy;
243 ev.stepDeltaX = stepDx;
244 ev.stepDeltaY = stepDy;
245 ev.durationMs = now - downTime_;
246 onGesture(ev);
247 }
248
249 void emitDiscrete(GestureType type, Point point, int16_t dx, int16_t dy,
250 uint32_t duration) {
251 if (!onGesture) return;
252 GestureEvent ev;
253 ev.type = type;
254 ev.phase = GesturePhase::kEnded;
255 ev.point = point;
256 ev.startPoint = downPoint_;
257 ev.deltaX = dx;
258 ev.deltaY = dy;
259 ev.durationMs = duration;
260 onGesture(ev);
261 }
262
263 /// Tracks a second touch point (if the driver ever supplies one) to
264 /// recognize pinch/rotate. See TouchDriver::getSecondPoint.
265 void updatePinchRotate(bool hasSecond, Point first, Point second) {
266 if (!hasSecond) {
267 secondActive_ = false;
268 return;
269 }
270
271 const float dxp = static_cast<float>(second.x - first.x);
272 const float dyp = static_cast<float>(second.y - first.y);
273 const float distance = sqrtf(dxp * dxp + dyp * dyp);
274 const float angle = atan2f(dyp, dxp) * 180.0f / 3.14159265f;
275
276 if (!secondActive_) {
277 secondActive_ = true;
278 initialPinchDistance_ = distance > 0.0f ? distance : 1.0f;
279 lastPinchDistance_ = distance;
280 initialPinchAngle_ = angle;
281 lastPinchAngle_ = angle;
282 return;
283 }
284
285 if (!onGesture) {
286 lastPinchDistance_ = distance;
287 lastPinchAngle_ = angle;
288 return;
289 }
290
291 const float distDelta = distance - lastPinchDistance_;
292 if (distDelta > 0.5f || distDelta < -0.5f) {
293 GestureEvent ev;
294 ev.type = (distDelta > 0.0f) ? GestureType::kPinchOut : GestureType::kPinchIn;
295 ev.phase = GesturePhase::kChanged;
296 ev.point = first;
297 ev.startPoint = downPoint_;
298 ev.scale = distance / initialPinchDistance_;
299 ev.durationMs = millis() - downTime_;
300 onGesture(ev);
301 }
302
303 float angleDelta = angle - lastPinchAngle_;
304 if (angleDelta > 180.0f) angleDelta -= 360.0f;
305 if (angleDelta < -180.0f) angleDelta += 360.0f;
306 if (angleDelta > 1.0f || angleDelta < -1.0f) {
307 GestureEvent ev;
308 ev.type = GestureType::kRotate;
309 ev.phase = GesturePhase::kChanged;
310 ev.point = first;
311 ev.startPoint = downPoint_;
312 ev.rotationDegrees = angle - initialPinchAngle_;
313 ev.durationMs = millis() - downTime_;
314 onGesture(ev);
315 }
316
317 lastPinchDistance_ = distance;
318 lastPinchAngle_ = angle;
319 }
320};
