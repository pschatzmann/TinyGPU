18/**
19 * @brief Hardware-accelerated IQSPIBus backend for ESP32, built on top of
20 * ESP-IDF's esp_lcd_new_panel_io_spi(quad_mode=1) - DMA-driven, with a
21 * persistent scratch buffer and completion callback for writeColor() (see
22 * that method's doc comment for why: esp_lcd_panel_io_tx_color() is
23 * asynchronous). This is a straight extraction of what used to live
24 * directly in DisplayDriverQSPI before the IQSPIBus split - behavior is
25 * unchanged from before.
26 */

89 bool writeColor(uint8_t cmd, const uint8_t* data, size_t len) override {
90 if (io_ == nullptr) return false;
91
92 // Copied into a scratch buffer because esp_lcd_panel_io_tx_color() is
93 // asynchronous (trans_queue_depth > 1 above): the caller's buffer
94 // could be reused/freed before DMA finishes reading it. Blocking (via
95 // on_color_trans_done) until the transfer completes turns this back
96 // into the synchronous call IQSPIBus::writeColor() promises.
97 if (scratchCapacity_ < len) {
98 delete[] scratch_;
99 scratch_ = new uint8_t[len];
100 scratchCapacity_ = len;
101 }
102 memcpy(scratch_, data, len);
103
104 colorTransPending_ = true;
105 if (esp_lcd_panel_io_tx_color(io_, qspiCmd(kOpcodeColor, cmd), scratch_,
106 len) != ESP_OK) {
107 colorTransPending_ = false;
108 return false;
109 }
110 while (colorTransPending_) {
111 // busy-wait for the DMA transfer to finish
112 }
113 return true;
114 }
115
116 private:

119
120 static int qspiCmd(uint8_t opcode, uint8_t cmd) {
121 return (static_cast<int>(opcode) << 24) | (static_cast<int>(cmd) << 8);
122 }

131 }
