39/// Milliseconds since the epoch, from a monotonic (steady) clock - not
40/// "time since boot" the way Arduino's millis() is, but that distinction
41/// never matters here: GestureDetector.h only ever subtracts two millis()
42/// calls to get an elapsed duration, never reads the absolute value.
43inline unsigned long millis() {
44 using namespace std::chrono;
45 return static_cast<unsigned long>(
46 duration_cast<milliseconds>(steady_clock::now().time_since_epoch())
47 .count());
48}
