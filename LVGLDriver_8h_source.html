11 * @brief LVGLDriver is a helper class to initialize LVGL v9 with a TinyGPU
12 * Output driver. It sets up the display buffer and flush callback for LVGL.
13 *
14 * @tparam RGB_T The pixel format LVGL's rendered output is converted to
15 * before transmission. Defaults to RGB565, matching LVGL's own internal
16 * RGB565 color format (convertBuffer() below still has to undo RGB565's
17 * own byte-swapped storage - see RGB565.h - even in that default case).
18 */

105 // Displays the given area of the screen with the provided color data.
106 static void display(lv_display_t* disp, const lv_area_t* area,
107 uint8_t* px_map) {
108 assert(disp != nullptr);
109 const size_t w = static_cast<size_t>(area->x2 - area->x1 + 1);
110 const size_t h = static_cast<size_t>(area->y2 - area->y1 + 1);
111
112 LVGLDriver* p_driver =
113 static_cast<LVGLDriver*>(lv_display_get_user_data(disp));
114
115 // px_map is packed by LVGL itself according to LV_COLOR_FORMAT_RGB565,
116 // which is LVGL's own fixed (native-order) bit layout - unrelated to
117 // RGB_T's own storage convention. Re-pack each pixel into RGB_T's
118 // layout (and apply gamma correction, if set) before handing the
119 // buffer to the driver.
120 convertBuffer(px_map, w * h);
121
122 // SurfaceWithExternalBuffer's default font argument is hardcoded to
123 // BitmapFont<RGB565>, which doesn't match IFont<RGB_T>& for any other
124 // RGB_T, so an explicit RGB_T-typed font has to be passed here.
125 static BitmapFont<RGB_T> font;
126 SurfaceWithExternalBuffer<RGB_T> surface(w, h, font);
127 surface.setExternalBuffer(px_map, w * h * sizeof(RGB_T));
128 surface.resizeBuffer(w, h);
129
130 p_driver->getDriver().writeData(surface, area->x1, area->y1);
131
132 lv_display_flush_ready(disp);
133 }

159 /// Applies independent per-channel gamma correction to every color LVGL
160 /// renders, before RGB_T's own field layout/compensation is applied.
161 /// gammaR=gammaG=gammaB=1.0 (the default) applies no correction.
162 ///
163 /// This is deliberately per-channel rather than one shared curve: some
164 /// panels show a visible color tint in near-black/grey tones even with
165 /// correct field compensation, because the panel's red/green/blue
166 /// subpixels have different non-linear responses at low brightness. A
167 /// single gamma applied equally to R, G and B can only ever change
168 /// overall brightness - if R=G=B going in, the same curve on all three
169 /// still gives R=G=B going out, so it mathematically cannot fix a hue
170 /// tint in grey tones. Independent per-channel curves can, by
171 /// deliberately un-balancing a grey input to counteract the panel's own
172 /// imbalance. There's no single correct set of values for an unknown/
173 /// undocumented panel - try values on both sides of 1.0 per channel and
174 /// compare on real hardware. See GammaTable.h.
175 static void setGamma(float gammaR, float gammaG, float gammaB) {
176 gammaR_().setGamma(gammaR);
177 gammaG_().setGamma(gammaG);
178 gammaB_().setGamma(gammaB);
179 }

206 /// Converts a buffer LVGL packed as (native-order) RGB565 into RGB_T's
207 /// own byte layout, applying per-channel gamma correction (see
208 /// setGamma()), in place. Each pixel is decoded with RGB565's semantics
209 /// (what LVGL actually wrote), gamma corrected, and re-encoded via
210 /// RGB_T::fromRGB(), so RGB_T's own compensation (if any) is genuinely
211 /// applied regardless of how RGB_T's constructor happens to order its
212 /// arguments.
213 static void convertBuffer(uint8_t* px_map, size_t pixelCount) {
214 const bool sameFormat = std::is_same<RGB_T, RGB565>::value;
215 const bool needsGamma = (gammaR_().gamma() != 1.0f) ||
216 (gammaG_().gamma() != 1.0f) ||
217 (gammaB_().gamma() != 1.0f);
218 uint16_t* pixels = reinterpret_cast<uint16_t*>(px_map);
219 if (sameFormat && !needsGamma) {
220 // No channel repacking or gamma to apply, but RGB565 stores bytes
221 // swapped from LVGL's native order (see RGB565.h) - a raw swap is
222 // still needed.
223 for (size_t i = 0; i < pixelCount; ++i) {
224 pixels[i] = RGB565::swapBytes(pixels[i]);
225 }
226 return;
227 }
228
229 const GammaTable& gr = gammaR_();
230 const GammaTable& gg = gammaG_();
231 const GammaTable& gb = gammaB_();
232 for (size_t i = 0; i < pixelCount; ++i) {
233 // pixels[i] is LVGL's native-order pixel; byte-swap it into
234 // RGB565's stored (wire) order before decoding.
235 const RGB565 src(RGB565::swapBytes(pixels[i]));
236 uint8_t r = src.getRed();
237 uint8_t g = src.getGreen();
238 uint8_t b = src.getBlue();
239 if (needsGamma) {
240 r = gr.apply(r);
241 g = gg.apply(g);
242 b = gb.apply(b);
243 }
244 const RGB_T dst = RGB_T::fromRGB(r, g, b);
245 pixels[i] = dst.getValue();
246 }
247 }

255 lv_indev_t* indev = lv_indev_create();
256 if (!indev) return nullptr;
257
258 lv_indev_set_type(indev, LV_INDEV_TYPE_POINTER);
259 lv_indev_set_display(indev, disp); // Binds touch to this specific display
260 lv_indev_set_user_data(indev, &touchDriver());
261
262 // Set static touch read callback
263 lv_indev_set_read_cb(indev, [](lv_indev_t* indev, lv_indev_data_t* data) {
264 auto* touch =
265 static_cast<TouchDriver*>(lv_indev_get_user_data(indev));
266 Point p;
267 if (touch && touch->getPoint(p)) {
268 data->point.x = p.x;
269 data->point.y = p.y;
270 data->state = LV_INDEV_STATE_PRESSED;
271 } else {
272 data->state = LV_INDEV_STATE_RELEASED;
273 }
274 });
275
276 return indev;
277 }
