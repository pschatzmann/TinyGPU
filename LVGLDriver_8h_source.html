12 * @brief LVGLDriver is a helper class to initialize LVGL v9 with a TinyGPU
13 * Output driver. It sets up the display buffer and flush callback for LVGL.
14 */
15class LVGLDriver {
16 public:
17 LVGLDriver(DisplayDriverSPI& driver, size_t x, size_t y,
18 size_t bufferSize = 0) {
19 this->driver = &driver;
20 this->disp_x = x;
21 this->disp_y = y;
22 // Default buffer size in bytes (v9 buffer size is expressed in bytes)
23 this->dispBufferSize =
24 bufferSize > 0 ? bufferSize : disp_x * sizeof(RGB565);
25 }

27 // Initializes LVGL and registers the TinyGPU Output driver with LVGL.
28 bool begin() {
29 if (!driver) {
30 return false;
31 }
32
33 lv_init();
34 lv_tick_set_cb(my_tick);
35
36 driver->begin();
37
38 if (touch_driver) {
39 CalibrationData cal;
40 cal.screenWidth = disp_x;
41 cal.screenHeight = disp_y;
42 touch_driver->setCalibration(cal);
43 touch_driver->begin();
44 setupTouch();
45 }
46
47 // Create display object in LVGL v9
48 disp = lv_display_create(disp_x, disp_y);
49 if (!disp) {
50 return false;
51 }
52 lv_display_set_color_format(disp, LV_COLOR_FORMAT_RGB565);
53
54 // Allocate draw buffer (byte vector for raw pixel memory)
55 buf_1.resize(dispBufferSize);
56
57 // Register display buffer with LVGL v9
58 lv_display_set_buffers(disp, buf_1.data(), nullptr, dispBufferSize,
59 LV_DISPLAY_RENDER_MODE_PARTIAL);
60
61 // Set user data and flush callback
62 lv_display_set_user_data(disp, this);
63 lv_display_set_flush_cb(disp, lvgl_tgpu_output_flush_cb);
64
65 return true;
66 }

68 void end() {
69 if (disp) {
70 lv_display_delete(disp);
71 disp = nullptr;
72 }
73 lv_tick_set_cb(nullptr);
74 lv_deinit();
75 buf_1.clear();
76 }

78 // Displays the given area of the screen with the provided color data.
79 static void display(lv_display_t* disp, const lv_area_t* area,
80 uint8_t* px_map) {
81 assert(disp != nullptr);
82 int32_t w = area->x2 - area->x1 + 1;
83 int32_t h = area->y2 - area->y1 + 1;
84 size_t bytes =
85 static_cast<size_t>(w) * static_cast<size_t>(h) * sizeof(RGB565);
86
87 LVGLDriver* p_driver =
88 static_cast<LVGLDriver*>(lv_display_get_user_data(disp));
89
90 SurfaceWithExternalBuffer<RGB565> surface(static_cast<size_t>(w),
91 static_cast<size_t>(h));
92
93 // 1. Assign external buffer (pass size in bytes)
94 surface.setExternalBuffer(px_map, w * h * sizeof(RGB565));
95
96 // 2. Resize surface dimensions (this updates width_ and height_)
97 surface.resizeBuffer(w, h);
98
99 // 3. Draw the surface to the TinyGPU Output
100 p_driver->getDriver().writeData(surface, area->x1, area->y1);
101
102 // 4. Notify LVGL
103 lv_display_flush_ready(disp);
104 }

138 /**
139 * @brief Registers a touch driver with this LVGL display instance.
140 */
141 lv_indev_t* setupTouch() {
142 if (!disp) return nullptr;
143
144 lv_indev_t* indev = lv_indev_create();
145 if (!indev) return nullptr;
146
147 lv_indev_set_type(indev, LV_INDEV_TYPE_POINTER);
148 lv_indev_set_display(indev, disp); // Binds touch to this specific display
149 lv_indev_set_user_data(indev, &touchDriver());
150
151 // Set static touch read callback
152 lv_indev_set_read_cb(indev, [](lv_indev_t* indev, lv_indev_data_t* data) {
153 auto* touch =
154 static_cast<TouchDriver*>(lv_indev_get_user_data(indev));
155 Point p;
156 if (touch && touch->getPoint(p)) {
157 data->point.x = p.x;
158 data->point.y = p.y;
159 data->state = LV_INDEV_STATE_PRESSED;
160 } else {
161 data->state = LV_INDEV_STATE_RELEASED;
162 }
163 });
164
165 return indev;
166 }
