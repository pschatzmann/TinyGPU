14class LVGLDriver {
15 public:
16 LVGLDriver(DisplayDriverSPI& driver, size_t x, size_t y,
17 size_t bufferSize = 0) {
18 this->driver = &driver;
19 this->disp_x = x;
20 this->disp_y = y;
21 // Default buffer size in bytes (v9 buffer size is expressed in bytes)

26 bool begin() {

28 return false;
29 }
30
31 lv_init();

39 return false;
40 }
41 lv_display_set_color_format(disp, LV_COLOR_FORMAT_RGB565);
42
43 // Allocate draw buffer (byte vector for raw pixel memory)
44 buf_1.resize(dispBufferSize);
45
46 // Register display buffer with LVGL v9
47 lv_display_set_buffers(disp, buf_1.data(), nullptr, dispBufferSize,

49
50 // Set user data and flush callback
51 lv_display_set_user_data(disp, this);
52 lv_display_set_flush_cb(disp, lvgl_tgpu_output_flush_cb);

55 }

67 // Displays the given area of the screen with the provided color data.
68 static void display(lv_display_t* disp, const lv_area_t* area,
69 uint8_t* px_map) {
70 assert(disp != nullptr);
71 int32_t w = area->x2 - area->x1 + 1;
72 int32_t h = area->y2 - area->y1 + 1;
73 size_t bytes =
74 static_cast<size_t>(w) * static_cast<size_t>(h) * sizeof(RGB565);
75
76 LVGLDriver* p_driver =
77 static_cast<LVGLDriver*>(lv_display_get_user_data(disp));
78
79 SurfaceWithExternalBuffer<RGB565> surface(static_cast<size_t>(w), static_cast<size_t>(h));
80
81 // 1. Assign external buffer (pass size in bytes)
82 surface.setExternalBuffer(px_map, w * h * sizeof(RGB565));
83
84 // 2. Resize surface dimensions (this updates width_ and height_)
85 surface.resizeBuffer(w, h);
86
87 // 3. Draw the surface to the TinyGPU Output
88 p_driver->getDriver().writeData(surface, area->x1, area->y1);
89
90 // 4. Notify LVGL
91 lv_display_flush_ready(disp);
92 }
