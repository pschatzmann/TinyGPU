15/**
16 * @brief PNG decoder for TinyGPU surfaces.
17 *
18 * Decodes a PNG file held in memory into an ISurface, which is resized to the
19 * image size. All colour types (grayscale, RGB, palette, with or without alpha)
20 * and bit depths 1, 2, 4, 8 and 16 are supported. Transparent pixels are mixed
21 * with the background colour. Interlaced (Adam7) images are not supported.
22 *
23 * The decoded data is kept in memory while decoding (the decompressed image),
24 * so the parser needs about (height * width * channels) bytes of RAM.
25 *
26 * @code
27 * Surface<RGB565> image(1, 1, FontRGB565);
28 * PNGParser<RGB565> parser(image);
29 * if (parser.decode(data, size)) { ... drawSprite(..., image) ... }
30 * @endcode
31 */
32template <typename RGB_T = RGB565>
33class PNGParser {
34 public:
35 /// Creates a parser that decodes into target.
36 explicit PNGParser(ISurface<RGB_T>& target) : target_(target) {}

38 /// Decodes a PNG file. Returns false on error (see errorMessage()).
39 bool decode(const uint8_t* data, size_t size, codec::Rgb8 background = codec::Rgb8(255, 255, 255)) {
40 error_ = nullptr;
41 width_ = 0;
42 height_ = 0;
43 if (data == nullptr || size < 8 || !hasSignature(data)) return fail("not a PNG file");
44
45 Header hdr;
46 std::vector<uint8_t> palette; // RGB triples
47 std::vector<uint8_t> alpha; // palette transparency
48 std::vector<uint8_t> idat;
49 size_t pos = 8;
50 bool haveHeader = false;
51 bool haveEnd = false;
52 while (pos + 12 <= size && !haveEnd) {
53 const uint32_t len = be32(data + pos);
54 const char* type = reinterpret_cast<const char*>(data + pos + 4);
55 if (len > size - pos - 12) return fail("truncated chunk");
56 const uint8_t* body = data + pos + 8;
57 if (memcmp(type, "IHDR", 4) == 0) {
58 if (len != 13) return fail("bad IHDR");
59 hdr.width = be32(body);
60 hdr.height = be32(body + 4);
61 hdr.depth = body[8];
62 hdr.colorType = body[9];
63 hdr.interlace = body[12];
64 if (body[10] != 0 || body[11] != 0) return fail("unknown compression or filter method");
65 haveHeader = true;
66 } else if (memcmp(type, "PLTE", 4) == 0) {
67 if (len % 3 != 0 || len > 256 * 3) return fail("bad PLTE");
68 palette.assign(body, body + len);
69 } else if (memcmp(type, "tRNS", 4) == 0) {
70 alpha.assign(body, body + len);
71 } else if (memcmp(type, "IDAT", 4) == 0) {
72 idat.insert(idat.end(), body, body + len);
73 } else if (memcmp(type, "IEND", 4) == 0) {
74 haveEnd = true;
75 }
76 pos += 12 + len;
77 }
78 if (!haveHeader) return fail("missing IHDR");
79 if (idat.empty()) return fail("missing IDAT");
80 if (hdr.width == 0 || hdr.height == 0) return fail("empty image");
81 if (hdr.interlace != 0) return fail("interlaced PNG is not supported");
82
83 const int channels = channelsOf(hdr.colorType);
84 if (channels == 0 || !validDepth(hdr.colorType, hdr.depth)) return fail("unsupported colour type or depth");
85 if (hdr.colorType == 3 && palette.empty()) return fail("palette missing");
86
87 const size_t bitsPerPixel = static_cast<size_t>(channels) * hdr.depth;
88 const size_t rowBytes = (static_cast<size_t>(hdr.width) * bitsPerPixel + 7) / 8;
89 const size_t bpp = bitsPerPixel < 8 ? 1 : bitsPerPixel / 8;
90 const size_t expected = static_cast<size_t>(hdr.height) * (rowBytes + 1);
91
92 std::vector<uint8_t> raw;
93 if (!codec::Inflate::zlib(idat.data(), idat.size(), raw, expected) || raw.size() < expected) {
94 return fail("corrupt image data");
95 }
96 if (!target_.resize(hdr.width, hdr.height)) return fail("cannot resize the target");
97
98 std::vector<uint8_t> prev(rowBytes, 0), cur(rowBytes, 0);
99 for (uint32_t y = 0; y < hdr.height; ++y) {
100 const uint8_t* line = raw.data() + y * (rowBytes + 1);
101 if (!unfilter(line[0], line + 1, prev, cur, bpp)) return fail("bad filter type");
102 for (uint32_t x = 0; x < hdr.width; ++x) {
103 target_.setPixel(x, y, codec::toColor<RGB_T>(pixel(hdr, cur, x, palette, alpha, background)));
104 }
105 prev.swap(cur);
106 }
107 width_ = hdr.width;
108 height_ = hdr.height;
109 return true;
110 }

120 static bool getSize(const uint8_t* data, size_t size, size_t& width, size_t& height) {
121 if (data == nullptr || size < 24 || !hasSignature(data)) return false;
122 if (memcmp(data + 12, "IHDR", 4) != 0) return false;
123 width = be32(data + 16);
124 height = be32(data + 20);

137 static bool hasSignature(const uint8_t* p) {
138 static const uint8_t kSignature[8] = {0x89, 'P', 'N', 'G', 0x0D, 0x0A, 0x1A, 0x0A};

142 static uint32_t be32(const uint8_t* p) {
143 return (static_cast<uint32_t>(p[0]) << 24) | (static_cast<uint32_t>(p[1]) << 16) |
144 (static_cast<uint32_t>(p[2]) << 8) | static_cast<uint32_t>(p[3]);
145 }

148 switch (colorType) {
149 case 0: return 1; // grayscale
150 case 2: return 3; // RGB

161 case 3: return depth == 1 || depth == 2 || depth == 4 || depth == 8;

163 case 4:

169 static uint8_t paeth(uint8_t a, uint8_t b, uint8_t c) {
170 const int p = static_cast<int>(a) + b - c;
171 const int pa = p > a ? p - a : a - p;
172 const int pb = p > b ? p - b : b - p;
173 const int pc = p > c ? p - c : c - p;
174 if (pa <= pb && pa <= pc) return a;
175 if (pb <= pc) return b;
176 return c;
177 }

179 /// Reverses the row filter of one scan line into out.
180 static bool unfilter(uint8_t type, const uint8_t* in, const std::vector<uint8_t>& prev,
181 std::vector<uint8_t>& out, size_t bpp) {
182 const size_t n = out.size();
183 for (size_t i = 0; i < n; ++i) {
184 const uint8_t left = i >= bpp ? out[i - bpp] : 0;
185 const uint8_t up = prev[i];
186 const uint8_t upLeft = i >= bpp ? prev[i - bpp] : 0;
187 uint8_t predicted = 0;
188 switch (type) {
189 case 0: predicted = 0; break;
190 case 1: predicted = left; break;
191 case 2: predicted = up; break;
192 case 3: predicted = static_cast<uint8_t>((left + up) / 2); break;
193 case 4: predicted = paeth(left, up, upLeft); break;
194 default: return false;
195 }
196 out[i] = static_cast<uint8_t>(in[i] + predicted);
197 }
198 return true;
199 }

201 /// The sample with the given index in a scan line, scaled to 0..255 unless it is a palette index.
202 static uint8_t sample(const std::vector<uint8_t>& line, size_t index, uint8_t depth, bool scale) {
203 if (depth == 16) return line[index * 2];
204 if (depth == 8) return line[index];
205 const size_t bit = index * depth;
206 const uint8_t value = static_cast<uint8_t>((line[bit / 8] >> (8 - depth - bit % 8)) & ((1u << depth) - 1));
207 if (!scale) return value;
208 return static_cast<uint8_t>(value * 255 / ((1u << depth) - 1));
209 }

211 static codec::Rgb8 pixel(const Header& hdr, const std::vector<uint8_t>& line, uint32_t x,
212 const std::vector<uint8_t>& palette, const std::vector<uint8_t>& alpha,
213 codec::Rgb8 background) {
214 const size_t channels = static_cast<size_t>(channelsOf(hdr.colorType));
215 const size_t first = static_cast<size_t>(x) * channels;
216 codec::Rgb8 c;
217 uint8_t a = 255;
218 switch (hdr.colorType) {
219 case 0: { // grayscale
220 const uint8_t g = sample(line, first, hdr.depth, true);
221 c = codec::Rgb8(g, g, g);
222 break;
223 }
224 case 4: { // grayscale + alpha
225 const uint8_t g = sample(line, first, hdr.depth, true);
226 c = codec::Rgb8(g, g, g);
227 a = sample(line, first + 1, hdr.depth, true);
228 break;
229 }
230 case 2: // RGB
231 c = codec::Rgb8(sample(line, first, hdr.depth, false), sample(line, first + 1, hdr.depth, false),
232 sample(line, first + 2, hdr.depth, false));
233 break;
234 case 6: // RGBA
235 c = codec::Rgb8(sample(line, first, hdr.depth, false), sample(line, first + 1, hdr.depth, false),
236 sample(line, first + 2, hdr.depth, false));
237 a = sample(line, first + 3, hdr.depth, false);
238 break;
239 case 3: { // palette
240 const size_t index = sample(line, first, hdr.depth, false);
241 if ((index + 1) * 3 <= palette.size()) {
242 c = codec::Rgb8(palette[index * 3], palette[index * 3 + 1], palette[index * 3 + 2]);
243 }
244 if (index < alpha.size()) a = alpha[index];
245 break;
246 }
247 default: break;
248 }
249 return codec::composite(c, background, a);
250 }
