15/***
16 * @brief Sprite Display without FrameBuffer. Usually we use a Framebuffer to
17 * compose and display sprites, but in some cases we want to display a sprite
18 * directly on the screen without using a framebuffer to minimize the required
19 * memory.
20 * The Sprites are also supporting touch events and can be moved, scaled and
21 * rotated.
22 * @tparam RGB_T The pixel format, default is RGB565.
23 */
24

42

62 /// Adds a sprite with a preallocated max buffer size for transformations.
63 SpriteInfo<RGB_T, Surface<RGB_T>>& addSprite(size_t x, size_t y, size_t maxX,
64 size_t maxY,
65 RGB_T backgroundColor) {
66 auto sprite = std::make_unique<Sprite<RGB_T>>(maxX, maxY, defaultFont_);
67 sprite->begin();
68 sprite->clear(backgroundColor);
69
70 // Pass color during addSprite or rely on constructor initialization
71 SpriteInfo<RGB_T, Surface<RGB_T>>& result = addSprite(x, y, *sprite);
72 result.isSurfaceAutoDelete = true;
73 // Ownership of the Sprite now belongs to the SpriteInfo (deleted via
74 // isSurfaceAutoDelete), so release it here to avoid a double free.
75 sprite.release();
76 return result;
77 }

79 /// Adds a sprite with a preallocated max buffer size for transformations.

95 /// Removes a sprite and restores the pixels behind it.
96 void removeSprite(SpriteInfo<RGB_T, Surface<RGB_T>>& sprite) {
97 TinyGPULogger.log(TinyGPULoggerClass::INFO, "Removing sprite at (%zu, %zu)",
98 sprite.x, sprite.y);
99
100 auto it = tinygpu::findSprite(sprites_, sprite);
101 if (it != sprites_.end()) {
102 clearSprite(sprite.x, sprite.y, sprite.currentSprite());
103 sprites_.erase(it);
104 }
105 }

107 /// Returns a reference to the sprite at the given position, if any.
108 SpriteInfo<RGB_T, Surface<RGB_T>>* getSprite(size_t x, size_t y) {
109 for (const auto& sprite : sprites_) {
110 if (x >= sprite->x && x < sprite->x + sprite->currentSprite().width() &&
111 y >= sprite->y && y < sprite->y + sprite->currentSprite().height()) {
112 return sprite.get();

140 /// Moves a sprite to a new position and redraws it.

143 TinyGPULogger.log(TinyGPULoggerClass::INFO,

196 /// Processes touch input and dispatches events to sprites if touched.
197 void processTouch() {
198 if (p_touchDriver && p_touchDriver->isTouched()) {
199 Point point;
200 if (p_touchDriver->getPoint(point)) {
201 TinyGPULogger.log(TinyGPULoggerClass::INFO,
202 "Touch detected at (%d, %d) with pressure %d",
203 point.x, point.y, point.pressure);
204 SpriteInfo<RGB_T, Surface<RGB_T>>* sprite = getSprite(point.x, point.y);
205 if (sprite) {
206 TinyGPULogger.log(TinyGPULoggerClass::INFO,
207 "Touch is within sprite at (%zu, %zu)", sprite->x,
208 sprite->y);
209 sprite->onTouch(point);
210 }
211 }
212 }

227 /// Helper to calculate centered coordinate when scaling/rotating
228 static size_t centeredCoordinate(size_t pos, size_t oldDim, size_t newDim) {
229 if (newDim >= oldDim) {
230 size_t diff = (newDim - oldDim) / 2;
231 return (pos > diff) ? (pos - diff) : 0;
232 } else {
233 return pos + (oldDim - newDim) / 2;
234 }
235 }

237 /// Applies a transformed sprite image to the display and updates bookkeeping.
238 void applyTransformedSprite(SpriteInfo<RGB_T, Surface<RGB_T>>& spriteInfo,
239 Surface<RGB_T>&& transformedSprite) {
240 const Rect oldBounds{spriteInfo.x, spriteInfo.y,
241 spriteInfo.currentSprite().width(),
242 spriteInfo.currentSprite().height()};
243 const size_t anchoredX = centeredCoordinate(spriteInfo.x, oldBounds.width,
244 transformedSprite.width());
245 const size_t anchoredY = centeredCoordinate(spriteInfo.y, oldBounds.height,
246 transformedSprite.height());
247
248 clearSprite(spriteInfo.x, spriteInfo.y, oldBounds.width, oldBounds.height);
249
250 spriteInfo.x = anchoredX;
251 spriteInfo.y = anchoredY;
252 spriteInfo.setTransformedSprite(std::move(transformedSprite));
253 displaySprite(spriteInfo.x, spriteInfo.y, spriteInfo.currentSprite());
254 }

259 clearSprite(x, y, sprite.width(), sprite.height());

270 // Route via writeData instead of drawSurface
271 driver_.writeData(actual_sprite_background_, x, y);

275 // Route via writeData instead of drawSurface
276 driver_.writeData(const_cast<ISurface<RGB_T>&>(sprite), x, y);
277 }
