16/***
17 * @brief Sprite Display without FrameBuffer. Usually we use a Framebuffer to
18 * compose and display sprites, but in some cases we want to display a sprite
19 * directly on the screen without using a framebuffer to minimize the required
20 * memory.
21 * The Sprites are also supporting touch events and can be moved, scaled and
22 * rotated.
23 * @tparam RGB_T The pixel format, default is RGB565.
24 */
25

36 /// Constructs from an LCDBoard (see LCDBoards.h): takes its display
37 /// driver and dimensions, and auto-attaches its touch controller if one
38 /// is present. Only valid for RGB_T = RGB565, since every LCDBoard's
39 /// display() is RGB565.
40 ///
41 /// begin() on the resulting SpriteDisplay calls LCDBoard::begin()
42 /// (backlight + SPI/QSPI bus + display controller + touch controller)
43 /// instead of just the display driver's begin() - do not also call
44 /// board.begin() yourself, or the display controller's init sequence
45 /// (and, on boards that need it, the display-inversion fix) runs twice.
46 explicit SpriteDisplay(LCDBoard& board, RGB_T backgroundColor = RGB_T(0))
47 : driver_(board.display()),
48 p_touchDriver(board.touch()),
49 p_board_(&board),
50 width_(board.width()),
51 height_(board.height()),
52 backgroundColor_(backgroundColor) {}

54 /// Adds a sprite to the framebuffer and draws it at the given position.
55 SpriteInfo<RGB_T, Surface<RGB_T>>& addSprite(size_t x, size_t y,
56 const ISurface<RGB_T>& sprite) {
57 TinyGPULogger.log(TinyGPULoggerClass::INFO, "Adding sprite at (%zu, %zu)",
58 x, y);
59 auto info = std::make_unique<SpriteInfo<RGB_T, Surface<RGB_T>>>(
60 x, y, sprite, backgroundColor_, defaultFont_);
61
62 displaySprite(x, y, info->currentSprite());
63
64 sprites_.push_back(std::move(info));
65 return *sprites_.back();
66 }
67
68 /// Starts the processing. If constructed from an LCDBoard, this brings
69 /// up the whole board (LCDBoard::begin()); otherwise just the display
70 /// driver (driver_.begin()).
71 bool begin() {
72 bool ok = p_board_ != nullptr ? p_board_->begin() : driver_.begin();
73 clear();
74 return ok;
75 }
76
77 /// Ends the processing
78 void end() {
79 driver_.end();
80 clear();
81 }

83 /// Adds a sprite with a preallocated max buffer size for transformations.
84 SpriteInfo<RGB_T, Surface<RGB_T>>& addSprite(size_t x, size_t y, size_t maxX,
85 size_t maxY,
86 RGB_T backgroundColor) {
87 auto sprite = std::make_unique<Sprite<RGB_T>>(maxX, maxY, defaultFont_);
88 sprite->begin();
89 sprite->clear(backgroundColor);
90
91 // Pass color during addSprite or rely on constructor initialization
92 SpriteInfo<RGB_T, Surface<RGB_T>>& result = addSprite(x, y, *sprite);
93 result.isSurfaceAutoDelete = true;
94 // Ownership of the Sprite now belongs to the SpriteInfo (deleted via
95 // isSurfaceAutoDelete), so release it here to avoid a double free.
96 sprite.release();
97 return result;
98 }

100 /// Adds a sprite with a preallocated max buffer size for transformations.
101 SpriteInfo<RGB_T, Surface<RGB_T>>& addSprite(size_t x, size_t y, size_t maxX,
102 size_t maxY,
103 const ISurface<RGB_T>& sprite) {

116 /// Removes a sprite and restores the pixels behind it.
117 void removeSprite(SpriteInfo<RGB_T, Surface<RGB_T>>& sprite) {
118 TinyGPULogger.log(TinyGPULoggerClass::INFO, "Removing sprite at (%zu, %zu)",
119 sprite.x, sprite.y);
120
121 auto it = tinygpu::findSprite(sprites_, sprite);
122 if (it != sprites_.end()) {
123 clearSprite(sprite.x, sprite.y, sprite.currentSprite());
124 sprites_.erase(it);
125 }
126 }

128 /// Returns a reference to the sprite at the given position, if any.
129 SpriteInfo<RGB_T, Surface<RGB_T>>* getSprite(size_t x, size_t y) {
130 for (const auto& sprite : sprites_) {
131 if (x >= sprite->x && x < sprite->x + sprite->currentSprite().width() &&
132 y >= sprite->y && y < sprite->y + sprite->currentSprite().height()) {
133 return sprite.get();
134 }
135 }
136 return nullptr;
137 }

161 /// Moves a sprite to a new position and redraws it.
162 void moveSprite(SpriteInfo<RGB_T, Surface<RGB_T>>& sprite, size_t newX,
163 size_t newY) {
164 TinyGPULogger.log(TinyGPULoggerClass::INFO,
165 "Moving sprite from (%zu, %zu) to (%zu, %zu)", sprite.x,
166 sprite.y, newX, newY);
167 if (sprite.x == newX && sprite.y == newY) {
168 return;
169 }
170
171 if (is_clear_on_sprite_move) {
172 clearSprite(sprite.x, sprite.y, sprite.currentSprite());
173 }
174
175 sprite.x = newX;
176 sprite.y = newY;
177 displaySprite(sprite.x, sprite.y, sprite.currentSprite());
178 }

224 /// Processes touch input and dispatches events to sprites if touched.
225 void processTouch() {
226 if (p_touchDriver && p_touchDriver->isTouched()) {
227 Point point;
228 if (p_touchDriver->getPoint(point)) {
229 TinyGPULogger.log(TinyGPULoggerClass::INFO,
230 "Touch detected at (%d, %d) with pressure %d",
231 point.x, point.y, point.pressure);
232 SpriteInfo<RGB_T, Surface<RGB_T>>* sprite = getSprite(point.x, point.y);
233 if (sprite) {
234 TinyGPULogger.log(TinyGPULoggerClass::INFO,
235 "Touch is within sprite at (%zu, %zu)", sprite->x,
236 sprite->y);
237 sprite->onTouch(point);
238 }
239 }
240 }

264 /// Helper to calculate centered coordinate when scaling/rotating
265 static size_t centeredCoordinate(size_t pos, size_t oldDim, size_t newDim) {
266 if (newDim >= oldDim) {
267 size_t diff = (newDim - oldDim) / 2;
268 return (pos > diff) ? (pos - diff) : 0;
269 } else {
270 return pos + (oldDim - newDim) / 2;
271 }
272 }

274 /// Applies a transformed sprite image to the display and updates bookkeeping.
275 void applyTransformedSprite(SpriteInfo<RGB_T, Surface<RGB_T>>& spriteInfo,
276 Surface<RGB_T>&& transformedSprite) {
277 const Rect oldBounds{spriteInfo.x, spriteInfo.y,
278 spriteInfo.currentSprite().width(),
279 spriteInfo.currentSprite().height()};
280 // See FrameBuffer::applyTransformedSprite() for why this must be the
281 // clamped (post-setTransformedSprite) size, not transformedSprite's
282 // own raw size.
283 const Rect clamped = spriteInfo.clampedSize(transformedSprite.width(),
284 transformedSprite.height());
285 const size_t anchoredX =
286 centeredCoordinate(spriteInfo.x, oldBounds.width, clamped.width);
287 const size_t anchoredY =
288 centeredCoordinate(spriteInfo.y, oldBounds.height, clamped.height);
289
290 clearSprite(spriteInfo.x, spriteInfo.y, oldBounds.width, oldBounds.height);
291
292 spriteInfo.x = anchoredX;
293 spriteInfo.y = anchoredY;
294 spriteInfo.setTransformedSprite(std::move(transformedSprite));
295 displaySprite(spriteInfo.x, spriteInfo.y, spriteInfo.currentSprite());
296 }

319 }
