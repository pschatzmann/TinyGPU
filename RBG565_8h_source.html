7/**
8 * @brief Color stored in a 16-bit word with an R:5 - B:6 - G:5 bit layout
9 * (green and blue swapped relative to RGB565), following the same
10 * constructor-order-matches-the-class-name convention as BGR565: pass
11 * components in R, B, G order and they're packed into those field
12 * positions directly, with no reordering.
13 *
14 * Some cheap ILI9341-compatible clone controllers don't honor the MADCTL
15 * BGR/RGB bit per the datasheet and instead always route the RGB565
16 * green-field position to the physical blue subpixel and the blue-field
17 * position to the physical green subpixel. On such panels, colors sent as
18 * standard RGB565 come out with green and blue swapped (e.g. a green fill
19 * shows up blue). Use RBG565 instead and place each color's intensity in
20 * the field that actually drives it on that hardware: red as normal, blue
21 * as the second argument, green as the third, e.g.
22 * RBG565 green(0, 0, 255); RBG565 blue(0, 255, 0);
23 */
24class RBG565 {
25 public:
26 /// Creates a black RBG565 color.
27 RBG565() = default;
28
29 /// Creates a color from 8-bit red, green, and blue components.
30 RBG565(uint8_t r, uint8_t b, uint8_t g) { setValue(r, b, g); }
31
32 /// Creates a color from a packed RBG565 value.
33 explicit RBG565(uint16_t packed) : value(packed) {}
34
35 /// Creates a color from semantic red/green/blue intensities, reordered
36 /// to this class's own (r, b, g) constructor convention internally.
37 /// Prefer this over the constructor in generic code that converts
38 /// between pixel formats, where callers work with plain r, g, b and
39 /// shouldn't need to know this class's field order.
40 static RBG565 fromRGB(uint8_t r, uint8_t g, uint8_t b) {
41 return RBG565(r, b, g);
42 }
43
44 /// Sets the color from 8-bit red, green, and blue components.
45 void setValue(uint8_t r, uint8_t b, uint8_t g) {
46 value = (static_cast<uint16_t>(r & 0xF8) << 8) |
47 (static_cast<uint16_t>(b & 0xFC) << 3) |
48 (static_cast<uint16_t>(g) >> 3);
49 }
50
51 /// Sets the color from a packed RBG565 value.
52 void setValue(uint16_t packed) { value = packed; }
53
54 /// Returns the red component expanded to 8 bits.
55 uint8_t getRed() const {
56 const uint8_t red = (value >> 11) & 0x1F;
57 return (red << 3) | (red >> 2);
58 }
59
60 /// Returns the green component expanded to 8 bits.
61 uint8_t getGreen() const {
62 const uint8_t green = value & 0x1F;
63 return (green << 3) | (green >> 2);
64 }
65
66 /// Returns the blue component expanded to 8 bits.
67 uint8_t getBlue() const {
68 const uint8_t blue = (value >> 5) & 0x3F;
69 return (blue << 2) | (blue >> 4);
70 }
71
72 /// Returns the packed RBG565 value.
73 uint16_t getValue() const { return value; }
74
75 /// Returns size in bits
76 static uint8_t size() { return 16; }
77
78 protected:
79 uint16_t value = 0;
80};
