110 /// Moves a sprite to a new position and redraws it.
111 void moveSprite(SpriteInfo<RGB_T, SurfaceT>& spriteInfo, size_t newX, size_t newY) {
112 TinyGPULogger.log(TinyGPULoggerClass::INFO,
113 "Moving sprite from (%zu, %zu) to (%zu, %zu)",
114 spriteInfo.x, spriteInfo.y, newX, newY);
115 if (spriteInfo.x == newX && spriteInfo.y == newY) {
116 return;
117 }
118
119 const Rect oldBounds{spriteInfo.x, spriteInfo.y,
120 spriteInfo.currentSprite().width(),
121 spriteInfo.currentSprite().height()};
122 const Rect newBounds{newX, newY, spriteInfo.currentSprite().width(),
123 spriteInfo.currentSprite().height()};
124 const Rect overlap = Rect::intersect(oldBounds, newBounds);
125
126 restoreExposedPixels(spriteInfo, oldBounds, overlap);
127 Surface<RGB_T> movedBackground =
128 captureUpdatedBackground(spriteInfo, oldBounds, newBounds, overlap);
129
130 spriteInfo.x = newX;
131 spriteInfo.y = newY;
132 spriteInfo.originalPixels = std::move(movedBackground);
133 surface_.drawSprite(spriteInfo.x, spriteInfo.y, spriteInfo.currentSprite(),
134 spriteInfo.invisibleColor);
135 }

253 /// Processes touch input and dispatches events to sprites if touched.
254 void processTouch() {
255 if (p_touchDriver && p_touchDriver->isTouched()) {
256 Point point;
257 if (p_touchDriver->getPoint(point)) {
258 TinyGPULogger.log(TinyGPULoggerClass::INFO,
259 "Touch detected at (%d, %d) with pressure %d",
260 point.x, point.y, point.pressure);
261 auto* sprite = getSprite(point.x, point.y);
262 if (sprite) {
263 TinyGPULogger.log(TinyGPULoggerClass::INFO,
264 "Touch is within sprite at (%zu, %zu)", sprite->x,
265 sprite->y);
266 // Call the sprite's onTouch callback if set
267 sprite->onTouch(point);
268 }
269 }
270 }

287 /// Applies a transformed sprite image to the framebuffer and updates
288 /// bookkeeping.
289 void applyTransformedSprite(SpriteInfo<RGB_T, SurfaceT>& spriteInfo,
290 Surface<RGB_T>&& transformedSprite) {
291 const Rect oldBounds{spriteInfo.x, spriteInfo.y,
292 spriteInfo.currentSprite().width(),
293 spriteInfo.currentSprite().height()};
294 const size_t anchoredX = centeredCoordinate(spriteInfo.x, oldBounds.width,
295 transformedSprite.width());
296 const size_t anchoredY = centeredCoordinate(spriteInfo.y, oldBounds.height,
297 transformedSprite.height());
298 const Rect newBounds{anchoredX, anchoredY, transformedSprite.width(),
299 transformedSprite.height()};
300 const Rect overlap = Rect::intersect(oldBounds, newBounds);
301
302 restoreExposedPixels(spriteInfo, oldBounds, overlap);
303 Surface<RGB_T> updatedBackground =
304 captureUpdatedBackground(spriteInfo, oldBounds, newBounds, overlap);
305
306 spriteInfo.x = anchoredX;
307 spriteInfo.y = anchoredY;
308 spriteInfo.setTransformedSprite(std::move(transformedSprite));
309 spriteInfo.originalPixels = std::move(updatedBackground);
310 this->drawSprite(spriteInfo.x, spriteInfo.y, spriteInfo.currentSprite(),
311 spriteInfo.invisibleColor);
312 }

315 /// Calculates the coordinate to center a new size over an old
316 /// position/size.
317 static size_t centeredCoordinate(size_t oldPosition, size_t oldSize,
318 size_t newSize) {
319 const float centeredPosition =
320 static_cast<float>(oldPosition) +
321 (static_cast<float>(oldSize) - static_cast<float>(newSize)) / 2.0;
322 return centeredPosition <= 0.0
323 ? 0
324 : static_cast<size_t>(std::lround(centeredPosition));
325 }

327 /// Restores only the pixels exposed by moving a sprite, using the overlap
328 /// region.
329 void restoreExposedPixels(const SpriteInfo<RGB_T, SurfaceT>& spriteInfo, const Rect& oldBounds,
330 const Rect& overlap) {
331 if (isEmpty(overlap)) {
332 restoreOriginalPixels(spriteInfo);
333 return;
334 }
335
336 drawSpriteRegion(spriteInfo.originalPixels, 0, 0, oldBounds.x, oldBounds.y,
337 oldBounds.width, overlap.y - oldBounds.y);
338
339 const size_t overlapBottom = overlap.y + overlap.height;
340 drawSpriteRegion(spriteInfo.originalPixels, 0, overlapBottom - oldBounds.y,
341 oldBounds.x, overlapBottom, oldBounds.width,
342 (oldBounds.y + oldBounds.height) - overlapBottom);
343
344 drawSpriteRegion(spriteInfo.originalPixels, 0, overlap.y - oldBounds.y,
345 oldBounds.x, overlap.y, overlap.x - oldBounds.x,
346 overlap.height);
347
348 const size_t overlapRight = overlap.x + overlap.width;
349 drawSpriteRegion(spriteInfo.originalPixels, overlapRight - oldBounds.x,
350 overlap.y - oldBounds.y, overlapRight, overlap.y,
351 (oldBounds.x + oldBounds.width) - overlapRight,
352 overlap.height);
353 }

355 /// Captures the updated background pixels after a sprite move/transform.
356 Surface<RGB_T> captureUpdatedBackground(const SpriteInfo<RGB_T, SurfaceT>& spriteInfo,
357 const Rect& oldBounds,
358 const Rect& newBounds,
359 const Rect& overlap) {
360 Surface<RGB_T> updatedBackground(newBounds.width, newBounds.height,
361 activeFont());
362 updatedBackground.begin();
363
364 if (isEmpty(overlap)) {
365 captureFramebufferRegion(updatedBackground, 0, 0, newBounds.x,
366 newBounds.y, newBounds.width, newBounds.height);
367 return updatedBackground;
368 }
369
370 copySpriteRegion(spriteInfo.originalPixels, overlap.x - oldBounds.x,
371 overlap.y - oldBounds.y, updatedBackground,
372 overlap.x - newBounds.x, overlap.y - newBounds.y,
373 overlap.width, overlap.height);
374
375 captureFramebufferRegion(updatedBackground, 0, 0, newBounds.x, newBounds.y,
376 newBounds.width, overlap.y - newBounds.y);
377
378 const size_t overlapBottom = overlap.y + overlap.height;
379 captureFramebufferRegion(updatedBackground, 0, overlapBottom - newBounds.y,
380 newBounds.x, overlapBottom, newBounds.width,
381 (newBounds.y + newBounds.height) - overlapBottom);
382
383 captureFramebufferRegion(updatedBackground, 0, overlap.y - newBounds.y,
384 newBounds.x, overlap.y, overlap.x - newBounds.x,
385 overlap.height);
386
387 const size_t overlapRight = overlap.x + overlap.width;
388 captureFramebufferRegion(updatedBackground, overlapRight - newBounds.x,
389 overlap.y - newBounds.y, overlapRight, overlap.y,
390 (newBounds.x + newBounds.width) - overlapRight,
391 overlap.height);
392
393 return updatedBackground;
394 }

396 /// Draws a rectangular region from a source surface onto the framebuffer.
397 void drawSpriteRegion(const ISurface<RGB_T>& source, size_t sourceX,
398 size_t sourceY, size_t destX, size_t destY,
399 size_t width, size_t height) {
400 for (size_t currentY = 0; currentY < height; ++currentY) {
401 for (size_t currentX = 0; currentX < width; ++currentX) {
402 const size_t framebufferX = destX + currentX;
403 const size_t framebufferY = destY + currentY;
404 if (framebufferX < surface_.width() &&
405 framebufferY < surface_.height()) {
406 surface_.setPixel(
407 framebufferX, framebufferY,
408 source.getPixel(sourceX + currentX, sourceY + currentY));
409 }
410 }
411 }
412 }

414 /// Copies a rectangular region from a source surface to a destination
415 /// surface.
416 static void copySpriteRegion(const ISurface<RGB_T>& source, size_t sourceX,
417 size_t sourceY, Surface<RGB_T>& destination,
418 size_t destX, size_t destY, size_t width,
419 size_t height) {
420 for (size_t currentY = 0; currentY < height; ++currentY) {
421 for (size_t currentX = 0; currentX < width; ++currentX) {
422 destination.setPixel(
423 destX + currentX, destY + currentY,
424 source.getPixel(sourceX + currentX, sourceY + currentY));
425 }
426 }
427 }

429 /// Captures a rectangular region from the framebuffer into a destination
430 /// surface.
431 void captureFramebufferRegion(Surface<RGB_T>& destination, size_t destX,
432 size_t destY, size_t sourceX, size_t sourceY,
433 size_t width, size_t height) {
434 for (size_t currentY = 0; currentY < height; ++currentY) {
435 for (size_t currentX = 0; currentX < width; ++currentX) {
436 const size_t framebufferX = sourceX + currentX;
437 const size_t framebufferY = sourceY + currentY;
438 const RGB_T color =
439 framebufferX < surface_.width() && framebufferY < surface_.height()
440 ? surface_.getPixel(framebufferX, framebufferY)
441 : RGB_T(0);
442 destination.setPixel(destX + currentX, destY + currentY, color);
443 }
444 }
445 }
