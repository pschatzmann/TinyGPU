13/**
14 * @brief Display class for rendering a Surface<RGB565> using the SDL2 library.
15 * This class can be used to display the contents of a TinyGPU Surface<RGB565>
16 * on a desktop environment for testing and debugging purposes.
17 *
18 * writeData() drains the SDL event queue on every call and exits the process
19 * on SDL_QUIT (i.e. the window's close button), since sketches that never
20 * touch a TouchDriverSDL - the only other place that pumps events - would
21 * otherwise leave the window unresponsive to being closed: with no one
22 * calling SDL_PollEvent, the window manager's close request just piles up in
23 * SDL's queue and the OS reports the window as hung.
24 *
25 * The window can be larger than the display: setScale() (or the environment
26 * variable TINYGPU_SDL_SCALE, which takes precedence) opens it with that
27 * factor, e.g. 1.5 or 2, and it can be resized - the display is scaled to the
28 * window (letterboxed if the aspect ratio differs): with sharp pixels for a
29 * whole factor, smoothed otherwise. TouchDriverSDL maps the mouse back to
30 * display coordinates.
31 */

79 static_assert(sizeof(RGB_T) == 2,

82 while (SDL_PollEvent(&event)) {

86 }
87 setAddressWindow(x, y, surface.width(), surface.height());
88 SDL_Rect updateRect = {static_cast<int>(x), static_cast<int>(y),
89 static_cast<int>(surface.width()),
90 static_cast<int>(surface.height())};
91 // RGB_T (RGB565) values are stored byte-swapped to SPI/QSPI wire
92 // order (see RGB565.h), but SDL_PIXELFORMAT_RGB565 expects the
93 // conventional (native) bit layout - swap back into a scratch
94 // buffer before uploading, rather than mutating the surface itself.
95 const size_t n = surface.size();
96 if (scratch_.size() < n) scratch_.resize(n);
97 const uint16_t* src = reinterpret_cast<const uint16_t*>(surface.data());
98 uint16_t* dst = reinterpret_cast<uint16_t*>(scratch_.data());
99 const size_t pixelCount = n / sizeof(RGB_T);
100 for (size_t i = 0; i < pixelCount; ++i) {
101 dst[i] = RGB565::swapBytes(src[i]);
102 }
103 SDL_UpdateTexture(texture_, &updateRect, scratch_.data(),
104 surface.width() * sizeof(RGB_T));
105 SDL_Rect fullRect = {0, 0, static_cast<int>(w_), static_cast<int>(h_)};
106 SDL_RenderClear(renderer_);
107 SDL_RenderCopy(renderer_, texture_, NULL, &fullRect);
108 SDL_RenderPresent(renderer_);
109 return true;
110 }
