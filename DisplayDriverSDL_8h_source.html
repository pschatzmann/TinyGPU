13/**
14 * @brief Display class for rendering a Surface<RGB565> using the SDL2 library.
15 * This class can be used to display the contents of a TinyGPU Surface<RGB565>
16 * on a desktop environment for testing and debugging purposes.
17 *
18 * writeData() drains the SDL event queue on every call and exits the process
19 * on SDL_QUIT (i.e. the window's close button), since sketches that never
20 * touch a TouchDriverSDL - the only other place that pumps events - would
21 * otherwise leave the window unresponsive to being closed: with no one
22 * calling SDL_PollEvent, the window manager's close request just piles up in
23 * SDL's queue and the OS reports the window as hung.
24 */

57 static_assert(sizeof(RGB_T) == 2,
58 "DisplayDriverSDL assumes a 16bpp RGB_T (RGB565)");
59 SDL_Event event;
60 while (SDL_PollEvent(&event)) {
61 if (event.type == SDL_QUIT) {
62 exit(0);
63 }
64 }
65 setAddressWindow(x, y, surface.width(), surface.height());
66 SDL_Rect updateRect = {static_cast<int>(x), static_cast<int>(y),
67 static_cast<int>(surface.width()),
68 static_cast<int>(surface.height())};
69 // RGB_T (RGB565) values are stored byte-swapped to SPI/QSPI wire
70 // order (see RGB565.h), but SDL_PIXELFORMAT_RGB565 expects the
71 // conventional (native) bit layout - swap back into a scratch
72 // buffer before uploading, rather than mutating the surface itself.
73 const size_t n = surface.size();
74 if (scratch_.size() < n) scratch_.resize(n);
75 const uint16_t* src = reinterpret_cast<const uint16_t*>(surface.data());
76 uint16_t* dst = reinterpret_cast<uint16_t*>(scratch_.data());
77 const size_t pixelCount = n / sizeof(RGB_T);
78 for (size_t i = 0; i < pixelCount; ++i) {
79 dst[i] = RGB565::swapBytes(src[i]);
80 }
81 SDL_UpdateTexture(texture_, &updateRect, scratch_.data(),
82 surface.width() * sizeof(RGB_T));
83 SDL_Rect fullRect = {0, 0, static_cast<int>(w_), static_cast<int>(h_)};
84 SDL_RenderClear(renderer_);
85 SDL_RenderCopy(renderer_, texture_, NULL, &fullRect);
86 SDL_RenderPresent(renderer_);
87 return true;
88 }
