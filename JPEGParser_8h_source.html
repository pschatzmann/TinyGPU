13/**
14 * @brief Decodes baseline JPEG images into a TinyGPU ISurface, using
15 * pschatzmann/TinyJPEG (https://github.com/pschatzmann/TinyJPEG) as the
16 * underlying decoder.
17 *
18 * TinyJPEG is a separate, optional Arduino/CMake library - install it
19 * alongside TinyGPU and `#include <TinyGPU/IO/JPEGParser.h>` explicitly
20 * (this header is not pulled in by TinyGPU.h) to opt in to JPEG support.
21 *
22 * TinyJPEG streams decoded pixels one MCU block at a time rather than
23 * requiring memory for the whole decoded frame, so decode() resizes the
24 * target surface once (from the JPEG's own header) and then just fans
25 * each block out to setPixel() as it arrives - no intermediate frame
26 * buffer beyond the target surface itself.
27 */
28template <typename RGB_T = RGB565>
29class JPEGParser {
30 public:
31 explicit JPEGParser(ISurface<RGB_T>& target) : target_(target) {
32 // TinyJPEG's default output is packed RGB565 in conventional (native)
33 // bit order; RGB565::RGB565(uint16_t) expects the byte-swapped wire
34 // order this library stores instead (see RGB565.h) - setSwapBytes(true)
35 // makes the decoder do that swap for us so onBlock() can construct
36 // RGB_T directly from each output pixel.
37 decoder_.setSwapBytes(true);
38 decoder_.setCallback(&JPEGParser::onBlock);
39 }
40
41 /// Sets the output reduction factor: 1 (full size), 2, 4, or 8.
42 void setScale(uint8_t scale) { decoder_.setJpgScale(scale); }
43
44 /// Decodes a JPEG held entirely in memory into the target surface at (0, 0).
45 /// Resizes the target surface to the JPEG's own dimensions first.
46 bool decode(const uint8_t* data, size_t size) {
47 uint16_t w = 0, h = 0;
48 if (decoder_.getJpgSize(&w, &h, data, size) != JDR_OK) {
49 return setError("Could not read JPEG header");
50 }
51 if (!target_.resize(w, h)) {
52 return setError("Could not resize target surface");
53 }
54 decoder_.setUserData(this);
55 return checkResult(decoder_.drawJpg(0, 0, data, size));
56 }
57
58 /// Decodes a JPEG read from any file-like object (Arduino's fs::File,
59 /// SD's File, or anything else exposing available()/read()/position()/
60 /// seek() - see TinyJPEG's docs/decoding.md) into the target surface at
61 /// (0, 0). Resizes the target surface to the JPEG's own dimensions
62 /// first. `file` is read from its current position and left wherever
63 /// the decode stopped; this method rewinds to that starting position
64 /// only between reading the header and decoding the pixels.
65 template <typename FileT>
66 bool decode(FileT& file) {
67 const size_t startPos = file.position();
68 uint16_t w = 0, h = 0;
69 if (decoder_.getJpgSize(&w, &h, file) != JDR_OK) {
70 return setError("Could not read JPEG header");
71 }
72 if (!target_.resize(w, h)) {
73 return setError("Could not resize target surface");
74 }
75 file.seek(startPos);
76 decoder_.setUserData(this);
77 return checkResult(decoder_.drawJpg(0, 0, file));
78 }
79
80 /// Returns the latest error message, if any.
81 const char* errorMessage() const { return errorMessage_; }
82
83 protected:
84 ISurface<RGB_T>& target_;
85 tinyjpeg::TinyJPEGDecoder decoder_;
86 const char* errorMessage_ = nullptr;
87
88 bool setError(const char* message) {
89 errorMessage_ = message;
90 return false;
91 }
92
93 bool checkResult(JRESULT result) {
94 if (result == JDR_OK) {
95 errorMessage_ = nullptr;
96 return true;
97 }
98 return setError("JPEG decode failed");
99 }
100
101 static bool onBlock(tinyjpeg::TinyJPEGDecoder& decoder, int16_t x, int16_t y,
102 uint16_t w, uint16_t h, uint16_t* data) {
103 auto* self = static_cast<JPEGParser*>(decoder.getUserData());
104 for (uint16_t row = 0; row < h; ++row) {
105 const size_t py = static_cast<size_t>(y) + row;
106 for (uint16_t col = 0; col < w; ++col) {
107 const size_t px = static_cast<size_t>(x) + col;
108 // Trailing MCU blocks commonly overhang the image's right/bottom
109 // edge (JPEG blocks are 8x8/16x16-aligned; the image itself
110 // usually isn't) - drop whatever falls outside the target.
111 if (self->target_.contains(px, py)) {
112 self->target_.setPixel(px, py, RGB_T(data[(row * w) + col]));
113 }
114 }
115 }
116 return true;
117 }
118};
