21/**
22 * @brief Base driver for QSPI ("4-wire quad SPI") TFT panels: 1 clock, 1
23 * chip-select, 4 data lines, no separate D/C pin. Used by a family of
24 * compact panel controllers (NV3041A, ST77916, CO5300, SH8601,
25 * AXS15231B, ...) that all speak a de-facto-standardized QSPI wire
26 * protocol, wrapping each command in a fixed 32-bit frame: the top byte
27 * is an opcode (0x02 = "write, parameter bytes follow over 1 line", 0x32
28 * = "write, color/pixel data follows over 4 lines"), the next byte is
29 * the actual MIPI-DCS-style command (0x11 sleep-out, 0x2A/0x2B/0x2C
30 * column/row/RAM-write, 0x36 MADCTL, ...), and the low 16 bits go
31 * unused. ESP-IDF's esp_lcd_new_panel_io_spi() builds that frame for
32 * you when configured with quad_mode=1, dc_gpio_num=-1, and
33 * lcd_cmd_bits=32 - this class only has to pass it the right `lcd_cmd`
34 * values via qspiCmd(), not build the 32-bit frame by hand. Subclasses
35 * implement begin() themselves (calling the inherited beginBus() for the
36 * shared SPI-bus/panel-IO setup, then their own chip-specific init
37 * register sequence), the same division of labor DisplayDriverSPI's
38 * setupPinsAndReset() uses for its SPI-panel subclasses.
39 */

74 bool writeData(ISurface<RGB_T>& surface, size_t x, size_t y) override {
75 static_assert(sizeof(RGB_T) == 2,
76 "writeData assumes a 16bpp RGB_T (RGB565) stored in "
77 "wire byte order");
78 if (io_ == nullptr) return false;
79 if (!setAddressWindow(x, y, surface.width(), surface.height())) {
80 return false;
81 }
82
83 // RGB_T (RGB565) values are stored in the surface already
84 // byte-swapped to this panel's wire order (see RGB565.h), so no
85 // per-pixel swap is needed here anymore. Still copied into a scratch
86 // buffer because esp_lcd_panel_io_tx_color() is asynchronous
87 // (trans_queue_depth > 1 above), so unlike DisplayDriverSPI's
88 // writeData() this can't reuse/free a scratch buffer based on
89 // assumptions about when a previous transfer happens to finish -
90 // that raced with the DMA still reading the old buffer and crashed
91 // on real hardware. Using one persistent, surface-sized scratch
92 // buffer, and blocking (via the on_color_trans_done callback set up
93 // in beginBus()) until the whole transfer completes before
94 // returning.
95 //
96 // IMPORTANT: send this as ONE tx_color() call, not chunked at this
97 // level. esp_lcd's own SPI backend (esp_lcd_panel_io_spi.c) already
98 // splits large transfers into multiple queued DMA transactions
99 // internally, sending the 0x2C RAM-write command only once up
100 // front and keeping CS active across the internal chunks - that's
101 // why the completion callback only fires after the true last
102 // chunk. Chunking manually at this level by calling tx_color()
103 // repeatedly re-sends the 0x2C command every time, which resets
104 // this panel's write cursor back to the top-left of the address
105 // window on each call - confirmed on real hardware to scramble the
106 // image far worse than not chunking at all.
107 const size_t n = surface.size();
108 if (scratchCapacity_ < n) {
109 delete[] scratch_;
110 scratch_ = new uint8_t[n];
111 scratchCapacity_ = n;
112 }
113 memcpy(scratch_, surface.data(), n);
114
115 colorTransPending_ = true;
116 if (esp_lcd_panel_io_tx_color(io_, qspiCmd(kOpcodeColor, kCmdRamWrite),
117 scratch_, n) != ESP_OK) {
118 colorTransPending_ = false;
119 return false;
120 }
121 while (colorTransPending_) {
122 // busy-wait for the DMA transfer to finish
123 }
124 return true;
125 }

132 static int qspiCmd(uint8_t opcode, uint8_t cmd) {

136 /// Sets up the SPI bus (quad data lines) and the panel-IO layer that
137 /// does the 32-bit command wrapping. Subclasses call this first thing
138 /// in their begin(), then send their chip's init register sequence via
139 /// writeCommand() before returning.
140 bool beginBus() {
141 spi_bus_config_t busCfg = {};
142 busCfg.data0_io_num = d0_;
143 busCfg.data1_io_num = d1_;
144 busCfg.sclk_io_num = sclk_;
145 busCfg.data2_io_num = d2_;
146 busCfg.data3_io_num = d3_;
147 busCfg.max_transfer_sz =
148 static_cast<int>(width_ * height_ * sizeof(RGB_T));
149
150 if (spi_bus_initialize(host_, &busCfg, SPI_DMA_CH_AUTO) != ESP_OK) {
151 return false;
152 }
153
154 esp_lcd_panel_io_spi_config_t ioCfg = {};
155 ioCfg.cs_gpio_num = cs_;
156 ioCfg.dc_gpio_num = -1;
157 ioCfg.spi_mode = 0;
158 ioCfg.pclk_hz = pclkHz_;
159 ioCfg.trans_queue_depth = 10;
160 ioCfg.lcd_cmd_bits = 32;
161 ioCfg.lcd_param_bits = 8;
162 ioCfg.flags.quad_mode = 1;
163 ioCfg.on_color_trans_done = &DisplayDriverQSPI<RGB_T>::onColorTransDone;
164 ioCfg.user_ctx = this;
165
166 return esp_lcd_new_panel_io_spi(static_cast<esp_lcd_spi_bus_handle_t>(host_),
167 &ioCfg, &io_) == ESP_OK;
168 }

176 bool setAddressWindow(size_t x, size_t y, size_t w, size_t h) override {
177 const uint8_t caset[4] = {
178 static_cast<uint8_t>(x >> 8), static_cast<uint8_t>(x & 0xFF),
179 static_cast<uint8_t>((x + w - 1) >> 8),
180 static_cast<uint8_t>((x + w - 1) & 0xFF)};
181 const uint8_t raset[4] = {
182 static_cast<uint8_t>(y >> 8), static_cast<uint8_t>(y & 0xFF),
183 static_cast<uint8_t>((y + h - 1) >> 8),
184 static_cast<uint8_t>((y + h - 1) & 0xFF)};
185 return writeCommand(0x2A, caset, 4) && writeCommand(0x2B, raset, 4);
186 }

209/**
210 * @brief Driver for the New Vision NV3041A QSPI TFT controller (480x272
211 * panels on Guition/Sunton JC4827W543-class ESP32-S3 boards).
212 *
213 * The full vendor register-unlock/gate/source-timing/gamma init
214 * sequence, confirmed against real hardware (ported from
215 * moononournation/Arduino_GFX's Arduino_NV3041A driver - an earlier
216 * minimal sleep-out/pixel-format/MADCTL/display-on sequence produced a
217 * rotated color channel mapping: red rendered as blue, green as red,
218 * blue as green). Register 0x3A (pixel format) uses a chip-specific
219 * encoding on this part, not the standard MIPI-DCS one: 0x01 selects
220 * 16bpp RGB565 here, not the commonly-documented 0x55.
221 */
222template <typename RGB_T = RGB565>
223class NV3041ADriver : public DisplayDriverQSPI<RGB_T> {
224 public:
225 using DisplayDriverQSPI<RGB_T>::beginBus;
226 using DisplayDriverQSPI<RGB_T>::writeCommand;
227
228 // 32MHz matches Arduino_GFX's Arduino_NV3041A driver, which documents
229 // it as this chip's maximum supported speed and runs clean at it on
230 // real hardware; this driver's original 40MHz default showed scattered
231 // pixel corruption on large transfers, consistent with the panel/
232 // wiring not reliably sustaining a faster clock.
233 NV3041ADriver(int8_t cs, int8_t sclk, int8_t d0, int8_t d1, int8_t d2,
234 int8_t d3, size_t width = 480, size_t height = 272,
235 uint32_t pclkHz = 32000000)
236 : DisplayDriverQSPI<RGB_T>(cs, sclk, d0, d1, d2, d3, width, height,
237 pclkHz) {}
238
239 bool begin() override {
240 static_assert(sizeof(RGB_T) == 2,
241 "NV3041ADriver assumes a 16bpp RGB_T (RGB565)");
242
243 if (!beginBus()) return false;
244
245 for (size_t i = 0; i < sizeof(kInitOps) / sizeof(kInitOps[0]); ++i) {
246 const uint8_t cmd = kInitOps[i][0];
247 const uint8_t data = kInitOps[i][1];
248 writeCommand(cmd, &data, 1);
249 if (cmd == 0x11) delay(120); // Sleep out
250 }
251 delay(100); // after display on (0x29, the last op in kInitOps)
252
253 // This panel is an IPS type: its default (uninverted) state and the
254 // vendor gamma table above only produce correct colors with display
255 // inversion turned ON - without this, colors come out wrong in a
256 // gamma-dependent, non-uniform way (not a clean photographic
257 // negative), which is why this was easy to miss.
258 writeCommand(0x21); // INVON
259
260 return true;
261 }

266 // meanings beyond 0x3A/0x36/0x11/0x29 (gate/source timing, gamma,
