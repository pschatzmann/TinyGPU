15/**
16 * @brief GIF decoder for TinyGPU surfaces (GIF87a and GIF89a).
17 *
18 * Decodes the first image of a GIF file held in memory into an ISurface. The
19 * surface is resized to the logical screen of the GIF. The area outside the
20 * first image, and its transparent pixels, get the background colour. Local and
21 * global palettes and interlaced images are supported. Animations show their
22 * first frame only.
23 *
24 * The compressed data of the image is kept in memory while decoding.
25 *
26 * @code
27 * Surface<RGB565> image(1, 1, FontRGB565);
28 * GIFParser<RGB565> parser(image);
29 * if (parser.decode(data, size)) { ... }
30 * @endcode
31 */
32template <typename RGB_T = RGB565>
33class GIFParser {
34 public:
35 /// Creates a parser that decodes into target.
36 explicit GIFParser(ISurface<RGB_T>& target) : target_(target) {}

38 /// Decodes the first image of a GIF file. Returns false on error (see errorMessage()).
39 bool decode(const uint8_t* data, size_t size, codec::Rgb8 background = codec::Rgb8(255, 255, 255)) {
40 error_ = nullptr;
41 width_ = 0;
42 height_ = 0;
43 if (data == nullptr || size < 13 || memcmp(data, "GIF8", 4) != 0) return fail("not a GIF file");
44 if (data[4] != '7' && data[4] != '9') return fail("unknown GIF version");
45 if (data[5] != 'a') return fail("unknown GIF version");
46
47 const uint32_t screenW = le16(data + 6);
48 const uint32_t screenH = le16(data + 8);
49 if (screenW == 0 || screenH == 0) return fail("empty image");
50 const uint8_t screenFlags = data[10];
51 size_t pos = 13;
52 std::vector<uint8_t> globalPalette;
53 if (screenFlags & 0x80) {
54 const size_t n = 3u * (size_t(1) << ((screenFlags & 7) + 1));
55 if (pos + n > size) return fail("truncated palette");
56 globalPalette.assign(data + pos, data + pos + n);
57 pos += n;
58 }
59
60 bool transparent = false;
61 uint8_t transparentIndex = 0;
62 while (pos < size) {
63 const uint8_t block = data[pos++];
64 if (block == 0x21) { // extension
65 if (pos >= size) break;
66 const uint8_t label = data[pos++];
67 if (label == 0xF9 && pos + 6 <= size && data[pos] >= 4) { // graphic control
68 transparent = (data[pos + 1] & 1) != 0;
69 transparentIndex = data[pos + 4];
70 }
71 if (!skipSubBlocks(data, size, pos)) return fail("truncated extension");
72 } else if (block == 0x2C) { // image descriptor
73 if (pos + 9 > size) return fail("truncated image descriptor");
74 const uint32_t left = le16(data + pos);
75 const uint32_t top = le16(data + pos + 2);
76 const uint32_t w = le16(data + pos + 4);
77 const uint32_t h = le16(data + pos + 6);
78 const uint8_t flags = data[pos + 8];
79 pos += 9;
80 std::vector<uint8_t> palette = globalPalette;
81 if (flags & 0x80) {
82 const size_t n = 3u * (size_t(1) << ((flags & 7) + 1));
83 if (pos + n > size) return fail("truncated palette");
84 palette.assign(data + pos, data + pos + n);
85 pos += n;
86 }
87 if (palette.empty()) return fail("palette missing");
88 if (pos >= size) return fail("truncated image");
89 const int minCode = data[pos++];
90 if (minCode < 2 || minCode > 8) return fail("bad LZW code size");
91 std::vector<uint8_t> compressed;
92 if (!collectSubBlocks(data, size, pos, compressed)) return fail("truncated image data");
93 std::vector<uint8_t> indices;
94 if (!lzw(compressed, minCode, indices, static_cast<size_t>(w) * h)) return fail("corrupt image data");
95 if (indices.size() < static_cast<size_t>(w) * h) return fail("missing image data");
96
97 if (!target_.resize(screenW, screenH)) return fail("cannot resize the target");
98 const RGB_T backgroundColor = codec::toColor<RGB_T>(background);
99 target_.clear(backgroundColor);
100 const std::vector<uint32_t> rows = rowOrder(h, (flags & 0x40) != 0);
101 for (uint32_t k = 0; k < h; ++k) {
102 const uint32_t y = top + rows[k];
103 if (y >= screenH) continue;
104 for (uint32_t x = 0; x < w; ++x) {
105 const uint32_t sx = left + x;
106 if (sx >= screenW) break;
107 const uint8_t index = indices[static_cast<size_t>(k) * w + x];
108 if (transparent && index == transparentIndex) continue;
109 codec::Rgb8 c;
110 if (static_cast<size_t>(index) * 3 + 2 < palette.size()) {
111 c = codec::Rgb8(palette[index * 3], palette[index * 3 + 1], palette[index * 3 + 2]);
112 }
113 target_.setPixel(sx, y, codec::toColor<RGB_T>(c));
114 }
115 }
116 width_ = screenW;
117 height_ = screenH;
118 return true;
119 } else {
120 break; // trailer (0x3B) or unknown data: no image
121 }
122 }
123 return fail("no image in the GIF file");
124 }

144 /// Skips the data sub-blocks (size byte, data, ..., 0 terminator).
145 static bool skipSubBlocks(const uint8_t* data, size_t size, size_t& pos) {
146 while (pos < size) {
147 const uint8_t n = data[pos++];
148 if (n == 0) return true;
149 if (pos + n > size) return false;
150 pos += n;
151 }
152 return false;
153 }

155 /// Copies the data sub-blocks into out.
156 static bool collectSubBlocks(const uint8_t* data, size_t size, size_t& pos, std::vector<uint8_t>& out) {
157 while (pos < size) {
158 const uint8_t n = data[pos++];
159 if (n == 0) return true;
160 if (pos + n > size) return false;
161 out.insert(out.end(), data + pos, data + pos + n);
162 pos += n;
163 }
164 return false;
165 }

167 /// The image row that the k-th stored row belongs to (interlaced images are stored in four passes).
168 static std::vector<uint32_t> rowOrder(uint32_t h, bool interlaced) {
169 std::vector<uint32_t> rows;
170 if (!interlaced) {
171 for (uint32_t y = 0; y < h; ++y) rows.push_back(y);
172 return rows;
173 }
174 static const uint8_t kStart[4] = {0, 4, 2, 1};
175 static const uint8_t kStep[4] = {8, 8, 4, 2};
176 for (int pass = 0; pass < 4; ++pass) {
177 for (uint32_t y = kStart[pass]; y < h; y += kStep[pass]) rows.push_back(y);
178 }
179 return rows;
180 }

182 /// LZW decoder for GIF images (variable code size, clear and end codes).
183 static bool lzw(const std::vector<uint8_t>& data, int minCode, std::vector<uint8_t>& out, size_t expected) {
184 const int clear = 1 << minCode;
185 const int end = clear + 1;
186 int codeSize = minCode + 1;
187 int next = end + 1;
188 std::vector<uint16_t> prefix(4096, 0);
189 std::vector<uint8_t> suffix(4096, 0);
190 uint8_t stack[4097];
191 int prev = -1;
192 uint32_t acc = 0;
193 int accBits = 0;
194 size_t p = 0;
195 out.clear();
196 out.reserve(expected);
197
198 for (;;) {
199 while (accBits < codeSize) {
200 if (p >= data.size()) return out.size() >= expected;
201 acc |= static_cast<uint32_t>(data[p++]) << accBits;
202 accBits += 8;
203 }
204 const int code = static_cast<int>(acc & ((1u << codeSize) - 1));
205 acc >>= codeSize;
206 accBits -= codeSize;
207
208 if (code == clear) {
209 codeSize = minCode + 1;
210 next = end + 1;
211 prev = -1;
212 continue;
213 }
214 if (code == end) break;
215 if (prev < 0) { // first code after a clear: must be a literal
216 if (code >= clear) return false;
217 out.push_back(static_cast<uint8_t>(code));
218 prev = code;
219 continue;
220 }
221 if (code > next) return false;
222
223 // the string of prev (plus its first byte when code is not yet in the table)
224 const int source = code < next ? code : prev;
225 auto firstByte = [&](int c) {
226 while (c >= clear) c = prefix[c];
227 return static_cast<uint8_t>(c);
228 };
229 const uint8_t first = firstByte(source);
230 int n = 0;
231 for (int c = source; c >= clear; c = prefix[c]) stack[n++] = suffix[c];
232 stack[n++] = firstByte(source); // the literal at the root of the chain
233 for (int i = n - 1; i >= 0; --i) out.push_back(stack[i]);
234 if (code == next) out.push_back(first); // KwKwK case
235
236 if (next < 4096) {
237 prefix[next] = static_cast<uint16_t>(prev);
238 suffix[next] = code < next ? firstByte(code) : first;
239 ++next;
240 if (next == (1 << codeSize) && codeSize < 12) ++codeSize;
241 }
242 prev = code;
243 if (out.size() >= expected) break;
244 }
245 return out.size() >= expected;
246 }
