27/**
28 * @brief Base driver for MIPI-DSI panels driven through ESP-IDF's DBI
29 * (command channel) + DPI (video-timing channel) split, the interface
30 * ESP32-P4 exposes for DSI LCDs. Unlike DisplayDriverSPI/DisplayDriverQSPI,
31 * a DPI panel has no address-window/RAMWR concept: `esp_lcd_new_panel_dpi()`
32 * allocates a screen-sized, DMA-scanned framebuffer up front, and
33 * `esp_lcd_panel_draw_bitmap()` writes straight into a rect of it - the
34 * continuously-running DPI bridge just scans whatever's currently there.
35 * `setAddressWindow()` is therefore a no-op here; writeData() addresses
36 * directly via that rect instead.
37 *
38 * Only a single frame buffer (`num_fbs = 1`) is used, so a write landing
39 * while that region happens to be mid-scan can tear - this driver doesn't
40 * attempt the double-buffer-plus-dirty-rect scheme some vendor BSPs use to
41 * avoid that (see guition-jc4880p4-bsp's board_p4.c for an example of that
42 * approach); same tradeoff DisplayDriverSPI/QSPI already accept for a
43 * torn/partial SPI transfer landing mid-refresh.
44 *
45 * Subclasses provide the panel-specific init register sequence (sent over
46 * the DBI command channel via writeCommand()) and this panel's exact DSI
47 * lane count/speed, DPI clock, and video timing - none of that is
48 * standardized across MIPI-DSI panels the way the QSPI command-wrapping
49 * convention is across QSPI panels, so unlike DisplayDriverQSPI there's no
50 * shared default here beyond the plumbing.
51 */

111 bool writeData(ISurface<RGB_T>& surface, size_t x, size_t y) override {
112 static_assert(sizeof(RGB_T) == 2,
113 "DisplayDriverDSI assumes a 16bpp RGB_T (RGB565)");
114 if (panel_ == nullptr) return false;
115
116 // RGB_T (RGB565) values are stored byte-swapped to SPI/QSPI wire order
117 // (see RGB565.h) - a DPI framebuffer instead wants the conventional
118 // (native) RGB565 bit layout, so swap back into a scratch buffer
119 // before handing it to esp_lcd, same as DisplayDriverSDL does and for
120 // the same reason (this is a plain memory framebuffer, not a
121 // command-driven SPI panel).
122 const size_t n = surface.size();
123 if (scratchCapacity_ < n) {
124 delete[] scratch_;
125 scratch_ = new uint8_t[n];
126 scratchCapacity_ = n;
127 }
128 const uint16_t* src = reinterpret_cast<const uint16_t*>(surface.data());
129 uint16_t* dst = reinterpret_cast<uint16_t*>(scratch_);
130 const size_t pixelCount = n / sizeof(RGB_T);
131 for (size_t i = 0; i < pixelCount; ++i) {
132 dst[i] = RGB565::swapBytes(src[i]);
133 }
134
135 return esp_lcd_panel_draw_bitmap(panel_, x, y, x + surface.width(),
136 y + surface.height(),
137 scratch_) == ESP_OK;
138 }

263/**
264 * @brief Driver for the Sitronix ST7701S MIPI-DSI TFT controller, as wired
265 * on the Guition JC4880P443C_I_W (ESP32-P4, 480x800 portrait, 4.3").
266 *
267 * Init register sequence, DSI lane config (2 lanes @ 500Mbps), DPI clock
268 * (34MHz) and video timing all transcribed verbatim from
269 * guition-jc4880p4-bsp's board_p4_st7701_init.h / board_p4.c
270 * (https://github.com/ultramcu/guition-jc4880p4-bsp), which documents them
271 * as VERIFIED-ON-HARDWARE on this exact board - this class only re-expresses
272 * that same recipe as a TinyGPU DisplayDriver instead of a bespoke BSP
273 * function, it does not add or change any of the underlying values. As with
274 * every other example in this repo's esp32-p4-480x800-guition-mipi-dsi-touch-display
275 * board directory, this file itself has not been run on real hardware.
276 *
277 * Register meanings beyond 0x11 (SLPOUT) / 0x29 (DISPON) are vendor-specific
278 * (ST7701 command-bank-switched extension registers) and undocumented
279 * beyond that source.
280 */

297 bool begin() override {
298 static_assert(sizeof(RGB_T) == 2, "ST7701Driver assumes a 16bpp RGB_T (RGB565)");
299
300 setVideoTiming(/*hsyncPulseWidth=*/12, /*hsyncBackPorch=*/42,
301 /*hsyncFrontPorch=*/42, /*vsyncPulseWidth=*/2,
302 /*vsyncBackPorch=*/8, /*vsyncFrontPorch=*/166);

304 if (!beginBus()) return false;

310 }
311
312 if (!finishInit()) return false;

315 }
